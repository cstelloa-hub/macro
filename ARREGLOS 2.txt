Private Sub VolcarData(wsMarcas As Worksheet, wsData As Worksheet)
    
    wsData.Cells.Clear
    
    ' AHORA SON 5 COLUMNAS (agregamos EEFF)
    wsData.Range("A1").Value = "Código SBS"
    wsData.Range("B1").Value = "Nombre Fondo"
    wsData.Range("C1").Value = "Día"
    wsData.Range("D1").Value = "EEFF"
    wsData.Range("E1").Value = "Var.Adj."
    wsData.Range("A1:E1").Font.Bold = True
    wsData.Range("A1:E1").Interior.Color = RGB(200, 215, 235)
    
    Dim ultCol As Long, ultFila As Long
    ultCol = wsMarcas.Cells(3, wsMarcas.Columns.Count).End(xlToLeft).Column
    If ultCol < 1 Then Exit Sub
    If ultCol > 500 Then ultCol = 500
    
    ultFila = wsMarcas.UsedRange.Rows.Count
    If ultFila > 1000 Then ultFila = 1000
    If ultFila < 5 Then ultFila = 5
    
    Dim arrMarcas As Variant
    arrMarcas = wsMarcas.Range(wsMarcas.Cells(1, 1), wsMarcas.Cells(ultFila, ultCol)).Value
    
    Dim colsInicio As Collection
    Set colsInicio = New Collection
    
    Dim j As Long
    For j = 1 To ultCol
        Dim h As String
        h = LCase(Trim(CStr(arrMarcas(3, j))))
        If h = "día" Or h = "dia" Then colsInicio.Add j
    Next j
    
    If colsInicio.Count = 0 Then Exit Sub
    
    Dim arrSalida() As Variant
    ReDim arrSalida(1 To 50000, 1 To 5)  ' 5 columnas ahora
    Dim filaSal As Long: filaSal = 0
    
    Dim b As Long
    For b = 1 To colsInicio.Count
        Dim colIni As Long, colFin As Long
        colIni = colsInicio(b)
        If b < colsInicio.Count Then
            colFin = colsInicio(b + 1) - 1
        Else
            colFin = ultCol
        End If
        
        Dim nombre As String, codigo As String
        nombre = Trim(CStr(arrMarcas(1, colIni)))
        codigo = Trim(CStr(arrMarcas(2, colIni)))
        
        ' Detectar columnas EEFF y Var.Adj. dentro del bloque
        Dim colVar As Long: colVar = 0
        Dim colEEFF As Long: colEEFF = 0
        For j = colIni To colFin
            Dim hh As String
            hh = LCase(Replace(Trim(CStr(arrMarcas(3, j))), " ", ""))
            Select Case hh
                Case "var.adj.", "var.adj", "varadj", "varadj.", "variaciónajustada", "variacionajustada"
                    colVar = j
                Case "eeff", "ee.ff.", "ee.ff", "ef.fi.", "eff", "estadosfinancieros"
                    colEEFF = j
            End Select
        Next j
        
        If codigo = "" Or colVar = 0 Then GoTo SiguienteBloque
        
        Dim rr As Long, vaciosSeguidos As Long
        vaciosSeguidos = 0
        rr = 4
        Do
            If rr > ultFila Then Exit Do
            
            Dim vFch As Variant
            vFch = arrMarcas(rr, colIni)
            
            If IsDate(vFch) Then
                vaciosSeguidos = 0
                Dim vVar As Variant
                vVar = arrMarcas(rr, colVar)
                If IsNumeric(vVar) Then
                    filaSal = filaSal + 1
                    arrSalida(filaSal, 1) = codigo
                    arrSalida(filaSal, 2) = nombre
                    arrSalida(filaSal, 3) = CDate(vFch)
                    
                    ' EEFF (puede estar vacío si no existe la columna)
                    If colEEFF > 0 Then
                        Dim vEEFF As Variant
                        vEEFF = arrMarcas(rr, colEEFF)
                        arrSalida(filaSal, 4) = vEEFF
                    Else
                        arrSalida(filaSal, 4) = ""
                    End If
                    
                    arrSalida(filaSal, 5) = CDbl(vVar)
                End If
            Else
                Dim txt As String
                txt = LCase(Trim(CStr(vFch)))
                If txt = "f1" Or txt = "cc" Or txt = "dd" Or txt = "nav" Or txt = "%" Then Exit Do
                If txt = "" Then
                    vaciosSeguidos = vaciosSeguidos + 1
                    If vaciosSeguidos > 30 Then Exit Do
                Else
                    Exit Do
                End If
            End If
            rr = rr + 1
        Loop
        
SiguienteBloque:
    Next b
    
    If filaSal > 0 Then
        Dim arrFinal() As Variant
        ReDim arrFinal(1 To filaSal, 1 To 5)
        Dim i As Long, kk As Long
        For i = 1 To filaSal
            For kk = 1 To 5
                arrFinal(i, kk) = arrSalida(i, kk)
            Next kk
        Next i
        
        wsData.Range("A2").Resize(filaSal, 5).Value = arrFinal
        wsData.Range("C2:C" & filaSal + 1).NumberFormat = "dd/mm/yyyy"
        wsData.Range("E2:E" & filaSal + 1).NumberFormat = "0.00%"
    End If
    
    wsData.Columns("A:E").AutoFit
End Sub









Private Sub ArmarCuadro(wsResumen As Worksheet, wsData As Worksheet, _
                        wsCuadro As Worksheet, tipo As String)
    
    Dim colNombre As Long, colCodigo As Long, filaInicio As Long
    If tipo = "LOCAL" Then
        colNombre = 1: colCodigo = 2: filaInicio = 3
    Else
        colNombre = 4: colCodigo = 5: filaInicio = 3
    End If
    
    Dim fondosPedidos As Object
    Set fondosPedidos = CreateObject("Scripting.Dictionary")
    Dim ordenPedidos As Collection
    Set ordenPedidos = New Collection
    
    Dim ultFila As Long, i As Long
    ultFila = wsResumen.Cells(wsResumen.Rows.Count, colCodigo).End(xlUp).Row
    
    For i = filaInicio To ultFila
        Dim cod As String, nom As String
        cod = Trim(CStr(wsResumen.Cells(i, colCodigo).Value))
        nom = Trim(CStr(wsResumen.Cells(i, colNombre).Value))
        If cod <> "" Then
            fondosPedidos(cod) = nom
            ordenPedidos.Add cod
        End If
    Next i
    
    wsCuadro.Cells.Clear
    
    If fondosPedidos.Count = 0 Then
        wsCuadro.Range("A1").Value = "No hay fondos cargados en Resumen para " & tipo
        Exit Sub
    End If
    
    Dim ultData As Long
    ultData = wsData.Cells(wsData.Rows.Count, 1).End(xlUp).Row
    
    If ultData < 2 Then
        wsCuadro.Range("A1").Value = "La hoja " & wsData.Name & " está vacía."
        Exit Sub
    End If
    
    ' Ahora leemos A:E (5 columnas) porque agregamos EEFF
    Dim arr As Variant
    arr = wsData.Range("A2:E" & ultData).Value
    
    Dim datos As Object, periodos As Object
    Set datos = CreateObject("Scripting.Dictionary")
    Set periodos = CreateObject("Scripting.Dictionary")
    
    Dim r As Long
    For r = 1 To UBound(arr, 1)
        Dim codR As String, fDia As Variant, vVar As Variant
        codR = Trim(CStr(arr(r, 1)))
        fDia = arr(r, 3)
        vVar = arr(r, 5)  ' Var.Adj. ahora está en columna 5
        
        If fondosPedidos.exists(codR) And IsDate(fDia) And IsNumeric(vVar) Then
            Dim periodo As String
            If tipo = "LOCAL" Then
                periodo = Format(CDate(fDia), "yyyy-mm")
            Else
                periodo = Year(CDate(fDia)) & "-Q" & ((Month(CDate(fDia)) - 1) \ 3 + 1)
            End If
            periodos(periodo) = 1
            
            Dim clave As String
            clave = codR & "|" & periodo
            
            Dim fNum As Double, varNum As Double
            fNum = CDbl(CDate(fDia))
            varNum = CDbl(vVar)
            
            If datos.exists(clave) Then
                Dim parts() As String
                parts = Split(datos(clave), "||")
                Dim fechaGuardada As Double, varGuardada As Double
                fechaGuardada = CDbl(parts(0))
                varGuardada = CDbl(parts(1))
                
                Dim reemplazar As Boolean: reemplazar = False
                If varGuardada = 0 And varNum <> 0 Then
                    reemplazar = True
                ElseIf varGuardada <> 0 And varNum = 0 Then
                    reemplazar = False
                Else
                    If fNum > fechaGuardada Then reemplazar = True
                End If
                
                If reemplazar Then
                    datos(clave) = CStr(fNum) & "||" & CStr(varNum)
                End If
            Else
                datos(clave) = CStr(fNum) & "||" & CStr(varNum)
            End If
        End If
    Next r
    
    If periodos.Count = 0 Then
        wsCuadro.Range("A1").Value = "No se encontraron datos para los códigos indicados en Resumen (" & tipo & ")."
        Exit Sub
    End If
    
    Dim listaPer() As String
    ReDim listaPer(0 To periodos.Count - 1)
    Dim idx2 As Long: idx2 = 0
    Dim k As Variant
    For Each k In periodos.keys
        listaPer(idx2) = CStr(k): idx2 = idx2 + 1
    Next k
    OrdenarArray listaPer
    
    wsCuadro.Range("A1").Value = "Nombre Fondo"
    wsCuadro.Range("B1").Value = "Código SBS"
    
    Dim c As Long
    For c = 0 To UBound(listaPer)
        Dim encab As String
        If tipo = "LOCAL" Then
            Dim pm() As String
            pm = Split(listaPer(c), "-")
            Dim fchTmp As Date
            fchTmp = DateSerial(CInt(pm(0)), CInt(pm(1)), 1)
            encab = Format(fchTmp, "mmm-yy")
            wsCuadro.Cells(1, 3 + c).NumberFormat = "@"
            wsCuadro.Cells(1, 3 + c).Value = encab
        Else
            Dim pq() As String
            pq = Split(listaPer(c), "-Q")
            encab = "Q" & pq(1) & "-" & pq(0)
            wsCuadro.Cells(1, 3 + c).NumberFormat = "@"
            wsCuadro.Cells(1, 3 + c).Value = encab
        End If
    Next c
    
    Dim fila As Long: fila = 2
    Dim kk As Variant
    For Each kk In ordenPedidos
        wsCuadro.Cells(fila, 1).Value = fondosPedidos(kk)
        wsCuadro.Cells(fila, 2).Value = kk
        
        For c = 0 To UBound(listaPer)
            Dim cl As String
            cl = kk & "|" & listaPer(c)
            If datos.exists(cl) Then
                Dim pp() As String
                pp = Split(datos(cl), "||")
                wsCuadro.Cells(fila, 3 + c).Value = CDbl(pp(1))
                wsCuadro.Cells(fila, 3 + c).NumberFormat = "0.00%"
            End If
        Next c
        
        fila = fila + 1
    Next kk
    
    With wsCuadro
        .Rows(1).Font.Bold = True
        .Range("A1:B1").Interior.Color = RGB(200, 215, 235)
        .Range(.Cells(1, 3), .Cells(1, 3 + UBound(listaPer))).Interior.Color = RGB(220, 230, 200)
        .Columns.AutoFit
    End With

End Sub









Private Sub GenerarGraficosPorTipo(wsBase As Worksheet, wsGraf As Worksheet, tipos As Object)
    
    Const MAX_POR_GRAFICO As Long = 7  ' Si hay más de 7, parte en dos
    
    Dim ultFila As Long
    ultFila = wsBase.Cells(wsBase.Rows.Count, 1).End(xlUp).Row
    
    Dim r As Long
    Dim posicionGrafico As Long: posicionGrafico = 1
    
    r = 1
    Do While r <= ultFila
        Dim titulo As String
        titulo = CStr(wsBase.Cells(r, 1).Value)
        
        If InStr(UCase(titulo), "RETORNO ACUMULADO") > 0 Then
            Dim filaEnc As Long: filaEnc = r + 1
            Dim filaInicioDatos As Long: filaInicioDatos = r + 2
            
            Dim ultCol As Long
            ultCol = wsBase.Cells(filaEnc, wsBase.Columns.Count).End(xlToLeft).Column
            
            Dim filaFinDatos As Long: filaFinDatos = filaInicioDatos
            Do While Trim(CStr(wsBase.Cells(filaFinDatos, 2).Value)) <> ""
                filaFinDatos = filaFinDatos + 1
                If filaFinDatos > ultFila Then Exit Do
            Loop
            filaFinDatos = filaFinDatos - 1
            
            If filaFinDatos >= filaInicioDatos Then
                ' Agrupar filas por tipo
                Dim porTipo As Object
                Set porTipo = CreateObject("Scripting.Dictionary")
                Dim ordenTipos As Collection
                Set ordenTipos = New Collection
                
                Dim rrr As Long
                For rrr = filaInicioDatos To filaFinDatos
                    Dim tipoF As String
                    tipoF = Trim(CStr(wsBase.Cells(rrr, 3).Value))
                    If tipoF = "" Then tipoF = "Sin clasificar"
                    If Not porTipo.exists(tipoF) Then
                        Set porTipo(tipoF) = New Collection
                        ordenTipos.Add tipoF
                    End If
                    porTipo(tipoF).Add rrr
                Next rrr
                
                Dim tipoIter As Variant
                For Each tipoIter In ordenTipos
                    Dim listaFilas As Collection
                    Set listaFilas = porTipo(CStr(tipoIter))
                    
                    ' Si hay más fondos que el máximo, ordenar por peso y partir
                    If listaFilas.Count > MAX_POR_GRAFICO Then
                        Dim listaOrdenada As Collection
                        Set listaOrdenada = OrdenarFilasPorPeso(wsBase, listaFilas)
                        
                        ' Partir en dos: mayor peso primero, menor peso después
                        Dim mitad As Long
                        mitad = (listaOrdenada.Count + 1) \ 2
                        
                        Dim listaTop As Collection, listaBottom As Collection
                        Set listaTop = New Collection
                        Set listaBottom = New Collection
                        
                        Dim idx As Long
                        For idx = 1 To listaOrdenada.Count
                            If idx <= mitad Then
                                listaTop.Add listaOrdenada(idx)
                            Else
                                listaBottom.Add listaOrdenada(idx)
                            End If
                        Next idx
                        
                        CrearGrafico wsBase, wsGraf, listaTop, filaEnc, ultCol, _
                                     titulo & " - " & CStr(tipoIter) & " (mayores)", posicionGrafico
                        posicionGrafico = posicionGrafico + 1
                        
                        CrearGrafico wsBase, wsGraf, listaBottom, filaEnc, ultCol, _
                                     titulo & " - " & CStr(tipoIter) & " (menores)", posicionGrafico
                        posicionGrafico = posicionGrafico + 1
                    Else
                        CrearGrafico wsBase, wsGraf, listaFilas, filaEnc, ultCol, _
                                     titulo & " - " & CStr(tipoIter), posicionGrafico
                        posicionGrafico = posicionGrafico + 1
                    End If
                Next tipoIter
            End If
            
            r = filaFinDatos + 1
        Else
            r = r + 1
        End If
    Loop
End Sub

'==================================================================
' Ordena una lista de filas por el peso Dic-25 (columna D), de mayor a menor
'==================================================================
Private Function OrdenarFilasPorPeso(wsBase As Worksheet, listaFilas As Collection) As Collection
    ' Pasar la collection a array para ordenar
    Dim arr() As Variant
    ReDim arr(1 To listaFilas.Count, 1 To 2)  ' columna 1: fila, columna 2: peso
    
    Dim i As Long
    For i = 1 To listaFilas.Count
        Dim filaActual As Long
        filaActual = CLng(listaFilas(i))
        arr(i, 1) = filaActual
        
        Dim pesoVal As Variant
        pesoVal = wsBase.Cells(filaActual, 4).Value  ' Peso Dic-25 está en col D
        If IsNumeric(pesoVal) Then
            arr(i, 2) = CDbl(pesoVal)
        Else
            arr(i, 2) = 0
        End If
    Next i
    
    ' Bubble sort descendente por peso
    Dim j As Long
    Dim tmpFila As Variant, tmpPeso As Variant
    For i = 1 To listaFilas.Count - 1
        For j = i + 1 To listaFilas.Count
            If arr(i, 2) < arr(j, 2) Then
                tmpFila = arr(i, 1): tmpPeso = arr(i, 2)
                arr(i, 1) = arr(j, 1): arr(i, 2) = arr(j, 2)
                arr(j, 1) = tmpFila: arr(j, 2) = tmpPeso
            End If
        Next j
    Next i
    
    ' Convertir de vuelta a Collection
    Dim resultado As Collection
    Set resultado = New Collection
    For i = 1 To listaFilas.Count
        resultado.Add CLng(arr(i, 1))
    Next i
    
    Set OrdenarFilasPorPeso = resultado
End Function




Private Function EscribirTiposBloque(wsDest As Worksheet, filaInicio As Long, _
                                      wsPxR As Worksheet, info As String, _
                                      mesSel As String) As Long
    Dim fila As Long: fila = filaInicio
    
    Dim partes() As String
    partes = Split(info, "|")
    Dim filaEnc As Long: filaEnc = CLng(partes(0))
    Dim colDic25 As Long: colDic25 = CLng(partes(1))
    Dim colUltMes As Long: colUltMes = CLng(partes(2))
    
    Dim subpartes() As String
    subpartes = Split(partes(3), ";")
    
    ' Detectar si la sección es trimestral o mensual mirando el primer encabezado después de Dic-25
    Dim esTrimestral As Boolean: esTrimestral = False
    If colUltMes > colDic25 Then
        Dim primerEncab As String
        primerEncab = LCase(Trim(CStr(wsPxR.Cells(filaEnc, colDic25 + 1).Value)))
        If InStr(primerEncab, "q") = 1 Then esTrimestral = True
    End If
    
    ' Buscar la columna del mes/trimestre seleccionado
    Dim colMesSel As Long: colMesSel = 0
    Dim c As Long
    
    If esTrimestral Then
        ' Convertir mes seleccionado a su trimestre correspondiente
        Dim trimestreObjetivo As String
        trimestreObjetivo = MesATrimestre(mesSel)
        
        ' Buscar exacto
        For c = colDic25 + 1 To colUltMes
            If LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value))) = LCase(trimestreObjetivo) Then
                colMesSel = c
                Exit For
            End If
        Next c
        
        ' Si no existe, usar el último trimestre <= objetivo
        If colMesSel = 0 Then
            Dim mejorCol As Long: mejorCol = 0
            For c = colDic25 + 1 To colUltMes
                Dim trEncab As String
                trEncab = Trim(CStr(wsPxR.Cells(filaEnc, c).Value))
                If trEncab <> "" Then
                    If TrimestreComparable(trEncab) <= TrimestreComparable(trimestreObjetivo) Then
                        mejorCol = c
                    End If
                End If
            Next c
            colMesSel = mejorCol
        End If
    Else
        ' Mensual: buscar directo
        For c = colDic25 To colUltMes
            If LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value))) = LCase(mesSel) Then
                colMesSel = c
                Exit For
            End If
        Next c
    End If
    
    Dim primerMesCol As Long: primerMesCol = colDic25 + 1
    
    Dim i As Long
    For i = 1 To UBound(subpartes)
        Dim subItem As String: subItem = subpartes(i)
        If subItem <> "" Then
            Dim sp() As String
            sp = Split(subItem, ":")
            If UBound(sp) >= 1 Then
                Dim tipo As String: tipo = sp(0)
                Dim filaTipoPxR As Long: filaTipoPxR = CLng(sp(1))
                
                wsDest.Cells(fila, 1).Value = "  " & tipo
                
                ' MTD: valor del mes/trimestre seleccionado
                Dim mtd As Double: mtd = 0
                If colMesSel > 0 Then
                    Dim vMtd As Variant
                    vMtd = wsPxR.Cells(filaTipoPxR, colMesSel).Value
                    If IsNumeric(vMtd) Then mtd = CDbl(vMtd)
                End If
                wsDest.Cells(fila, 2).Value = mtd
                wsDest.Cells(fila, 2).NumberFormat = "0.00%"
                
                ' YTD: producto compuesto desde primer mes/trimestre hasta el seleccionado
                Dim ytd As Double: ytd = 0
                If colMesSel > 0 And colMesSel >= primerMesCol Then
                    Dim acumulado As Double: acumulado = 1
                    Dim cc As Long
                    For cc = primerMesCol To colMesSel
                        Dim vRet As Variant
                        vRet = wsPxR.Cells(filaTipoPxR, cc).Value
                        If IsNumeric(vRet) Then
                            acumulado = acumulado * (1 + CDbl(vRet))
                        End If
                    Next cc
                    ytd = acumulado - 1
                End If
                wsDest.Cells(fila, 3).Value = ytd
                wsDest.Cells(fila, 3).NumberFormat = "0.00%"
                
                fila = fila + 1
            End If
        End If
    Next i
    
    EscribirTiposBloque = fila
End Function

'==================================================================
' Convierte un mes tipo "abr-26" al trimestre "Q2-2026"
'==================================================================
Private Function MesATrimestre(mesTxt As String) As String
    On Error Resume Next
    Dim fch As Date
    fch = DateValue("1-" & mesTxt)
    
    If Err.Number <> 0 Then
        Err.Clear
        MesATrimestre = ""
        Exit Function
    End If
    On Error GoTo 0
    
    Dim mes As Integer, anio As Integer
    mes = Month(fch)
    anio = Year(fch)
    
    Dim trimestre As Integer
    trimestre = ((mes - 1) \ 3) + 1
    
    MesATrimestre = "Q" & trimestre & "-" & anio
End Function

'==================================================================
' Convierte "Q1-2026" a número comparable (2026.1, 2026.2, etc.)
'==================================================================
Private Function TrimestreComparable(tr As String) As Double
    On Error Resume Next
    Dim partes() As String
    partes = Split(tr, "-")
    If UBound(partes) < 1 Then
        TrimestreComparable = 0
        Exit Function
    End If
    
    Dim qStr As String, anioStr As String
    qStr = partes(0)
    anioStr = partes(1)
    
    Dim qNum As Integer
    qNum = CInt(Replace(LCase(qStr), "q", ""))
    
    Dim anioNum As Integer
    anioNum = CInt(anioStr)
    
    TrimestreComparable = anioNum + (qNum / 10)
    
    If Err.Number <> 0 Then
        Err.Clear
        TrimestreComparable = 0
    End If
    On Error GoTo 0
End Function




Sub ProbarMesATrimestre()
    Dim msj As String
    msj = "Pruebas de MesATrimestre:" & vbCrLf & vbCrLf
    
    msj = msj & "'ene-26' -> '" & MesATrimestre("ene-26") & "'" & vbCrLf
    msj = msj & "'feb-26' -> '" & MesATrimestre("feb-26") & "'" & vbCrLf
    msj = msj & "'mar-26' -> '" & MesATrimestre("mar-26") & "'" & vbCrLf
    msj = msj & "'abr-26' -> '" & MesATrimestre("abr-26") & "'" & vbCrLf
    msj = msj & "'may-26' -> '" & MesATrimestre("may-26") & "'" & vbCrLf
    
    MsgBox msj, vbInformation, "Prueba"
End Sub





Sub DiagnosticoExtranjeroFinal()
    Dim wsPxR As Worksheet
    Set wsPxR = ThisWorkbook.Sheets("Pesos x Retornos")
    
    Dim wsResumen As Worksheet
    Set wsResumen = ThisWorkbook.Sheets("Resumen Total")
    
    Dim mesSel As String
    mesSel = CStr(wsResumen.Range("B1").Value)
    
    Dim msj As String
    msj = "Mes en B1: '" & mesSel & "'" & vbCrLf
    msj = msj & "Trimestre: '" & MesATrimestre(mesSel) & "'" & vbCrLf & vbCrLf
    
    Dim info As String
    info = LocalizarSeccionPxR(wsPxR, "EXTRANJERO", "Fondo 2")
    
    If info = "" Then
        msj = msj & "NO se encontró sección EXTRANJERO/Fondo 2"
        MsgBox msj, vbCritical
        Exit Sub
    End If
    
    msj = msj & "info devuelto: '" & info & "'" & vbCrLf & vbCrLf
    
    Dim partes() As String
    partes = Split(info, "|")
    
    msj = msj & "Total partes: " & UBound(partes) + 1 & vbCrLf
    
    Dim filaEnc As Long: filaEnc = 0
    Dim colDic25 As Long: colDic25 = 0
    Dim colUltMes As Long: colUltMes = 0
    
    On Error Resume Next
    filaEnc = CLng(partes(0))
    colDic25 = CLng(partes(1))
    colUltMes = CLng(partes(2))
    On Error GoTo 0
    
    msj = msj & "Fila encabezados: " & filaEnc & vbCrLf
    msj = msj & "Col Dic-25: " & colDic25 & vbCrLf
    msj = msj & "Col Ult: " & colUltMes & vbCrLf & vbCrLf
    
    If filaEnc = 0 Or colDic25 = 0 Or colUltMes = 0 Then
        msj = msj & "ERROR: algún valor es 0"
        MsgBox msj, vbCritical
        Exit Sub
    End If
    
    If colUltMes > 200 Then colUltMes = 50
    
    msj = msj & "Encabezados:" & vbCrLf
    Dim c As Long
    For c = colDic25 To colUltMes
        Dim val As String: val = ""
        On Error Resume Next
        val = CStr(wsPxR.Cells(filaEnc, c).Value)
        On Error GoTo 0
        msj = msj & "  Col " & c & " = '" & val & "'" & vbCrLf
    Next c
    
    MsgBox msj, vbInformation, "Diagnóstico"
End Sub



Private Function LocalizarSeccionPxR(wsPxR As Worksheet, _
                                      origenBuscar As String, _
                                      fondoBuscar As String) As String
    
    Dim ultFila As Long
    ultFila = wsPxR.Cells(wsPxR.Rows.Count, 1).End(xlUp).Row
    
    ' Buscar la sección de origen (LOCAL o EXTRANJERO)
    Dim filaOrigen As Long: filaOrigen = 0
    Dim r As Long
    For r = 1 To ultFila
        If UCase(Trim(CStr(wsPxR.Cells(r, 1).Value))) = UCase(origenBuscar) Then
            filaOrigen = r
            Exit For
        End If
    Next r
    
    If filaOrigen = 0 Then
        LocalizarSeccionPxR = ""
        Exit Function
    End If
    
    ' Buscar fin de la sección (siguiente LOCAL/EXTRANJERO)
    Dim filaFinSeccion As Long: filaFinSeccion = ultFila
    For r = filaOrigen + 1 To ultFila
        Dim val As String: val = UCase(Trim(CStr(wsPxR.Cells(r, 1).Value)))
        If val = "LOCAL" Or val = "EXTRANJERO" Then
            filaFinSeccion = r - 1
            Exit For
        End If
    Next r
    
    ' Buscar fondo dentro de la sección
    Dim filaFondo As Long: filaFondo = 0
    For r = filaOrigen + 1 To filaFinSeccion
        If LCase(Trim(CStr(wsPxR.Cells(r, 1).Value))) = LCase(fondoBuscar) Then
            filaFondo = r
            Exit For
        End If
    Next r
    
    If filaFondo = 0 Then
        LocalizarSeccionPxR = ""
        Exit Function
    End If
    
    ' Buscar el SIGUIENTE fondo dentro de la sección (para limitar el bloque actual)
    Dim filaFinBloque As Long: filaFinBloque = filaFinSeccion
    For r = filaFondo + 1 To filaFinSeccion
        Dim etqF As String: etqF = LCase(Trim(CStr(wsPxR.Cells(r, 1).Value)))
        If etqF = "fondo 2" Or etqF = "fondo 3" Then
            filaFinBloque = r - 1
            Exit For
        End If
    Next r
    
    Dim filaEnc As Long: filaEnc = filaFondo + 1
    
    Dim ultCol As Long
    ultCol = wsPxR.Cells(filaEnc, wsPxR.Columns.Count).End(xlToLeft).Column
    If ultCol < 5 Then ultCol = 10
    If ultCol > 50 Then ultCol = 50
    
    Dim colDic25 As Long: colDic25 = 0
    Dim c As Long
    For c = 5 To ultCol
        Dim h As String: h = LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value)))
        If h = "dic-25" Or h = "dic25" Then colDic25 = c
    Next c
    
    If colDic25 = 0 Then colDic25 = 5
    
    ' Buscar MTD SOLO dentro del bloque (entre filaEnc y filaFinBloque)
    Dim filaMTD As Long: filaMTD = 0
    For r = filaEnc + 1 To filaFinBloque
        If UCase(Trim(CStr(wsPxR.Cells(r, 1).Value))) = "MTD" Then
            filaMTD = r
            Exit For
        End If
    Next r
    
    If filaMTD = 0 Then
        LocalizarSeccionPxR = ""
        Exit Function
    End If
    
    ' Recolectar tipos SOLO dentro del bloque (después de MTD hasta filaFinBloque)
    Dim tiposInfo As String: tiposInfo = ""
    For r = filaMTD + 1 To filaFinBloque
        Dim etiqueta As String: etiqueta = Trim(CStr(wsPxR.Cells(r, 1).Value))
        If etiqueta = "" Then Exit For
        If LCase(etiqueta) = "fondo 2" Or LCase(etiqueta) = "fondo 3" Then Exit For
        If UCase(etiqueta) = "LOCAL" Or UCase(etiqueta) = "EXTRANJERO" Then Exit For
        tiposInfo = tiposInfo & ";" & etiqueta & ":" & r
    Next r
    
    LocalizarSeccionPxR = filaEnc & "|" & colDic25 & "|" & ultCol & "|" & filaMTD & tiposInfo
End Function
