Sub DiagnosticoRapido()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("Pesos")
    
    ' Buscar FONDO 2 SOLO en columnas A-Z y filas 1-150
    Dim filaF2 As Long: filaF2 = 0
    Dim colF2 As Long: colF2 = 0
    Dim r As Long, c As Long
    
    For r = 60 To 150
        For c = 1 To 26
            If UCase(Trim(CStr(ws.Cells(r, c).Value))) = "FONDO 2" Then
                filaF2 = r
                colF2 = c
                Exit For
            End If
        Next c
        If filaF2 > 0 Then Exit For
    Next r
    
    If filaF2 = 0 Then
        MsgBox "NO se encontró 'FONDO 2' en filas 60-150", vbCritical
        Exit Sub
    End If
    
    Dim msj As String
    msj = "FONDO 2 está en fila " & filaF2 & ", columna " & colF2 & vbCrLf & vbCrLf
    
    ' Mostrar las 3 filas siguientes, 7 columnas a la derecha
    Dim rr As Long, cc As Long
    For rr = filaF2 To filaF2 + 3
        msj = msj & "Fila " & rr & ": "
        For cc = colF2 To colF2 + 6
            Dim val As String
            val = CStr(ws.Cells(rr, cc).Value)
            If Len(val) > 12 Then val = Left(val, 12)
            msj = msj & "[" & val & "] "
        Next cc
        msj = msj & vbCrLf
    Next rr
    
    MsgBox msj, vbInformation, "Diagnóstico"
End Sub





'==================================================================
' MACRO 5: Resumen Total (respeta B1 si ya tiene mes válido)
'==================================================================
Sub GenerarResumenTotal()
    
    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    Dim wsPxR As Worksheet, wsCuadroLoc As Worksheet
    
    On Error Resume Next
    Set wsPxR = wb.Sheets("Pesos x Retornos")
    Set wsCuadroLoc = wb.Sheets("Cuadro Locales")
    On Error GoTo 0
    
    If wsPxR Is Nothing Then
        MsgBox "Falta hoja 'Pesos x Retornos'. Ejecuta primero GenerarPesosXRetornos.", vbCritical
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    ' Si ya existe Resumen Total y tiene un mes válido en B1, conservarlo
    Dim mesActual As String: mesActual = ""
    Dim wsExist As Worksheet
    On Error Resume Next
    Set wsExist = wb.Sheets("Resumen Total")
    On Error GoTo 0
    If Not wsExist Is Nothing Then
        mesActual = Trim(CStr(wsExist.Range("B1").Value))
    End If
    
    ' Construir lista de meses disponibles
    Dim listaMeses As String: listaMeses = ""
    Dim ultMes As String: ultMes = "mar-26"
    Dim mesesValidos As Object
    Set mesesValidos = CreateObject("Scripting.Dictionary")
    
    If Not wsCuadroLoc Is Nothing Then
        Dim ultColLoc As Long
        ultColLoc = wsCuadroLoc.Cells(1, wsCuadroLoc.Columns.Count).End(xlToLeft).Column
        Dim c As Long
        For c = 3 To ultColLoc
            Dim enc As String
            enc = Trim(CStr(wsCuadroLoc.Cells(1, c).Value))
            If enc <> "" Then
                ultMes = enc
                mesesValidos(LCase(enc)) = 1
                If listaMeses = "" Then
                    listaMeses = enc
                Else
                    listaMeses = listaMeses & "," & enc
                End If
            End If
        Next c
    End If
    
    ' Determinar mes a usar: si el actual es válido, mantenerlo; si no, usar último
    Dim mesUsar As String
    If mesActual <> "" And mesesValidos.exists(LCase(mesActual)) Then
        mesUsar = mesActual
    Else
        mesUsar = ultMes
    End If
    
    Dim wsDest As Worksheet
    Set wsDest = ObtenerHoja(wb, "Resumen Total")
    wsDest.Cells.Clear
    
    wsDest.Range("A1").Value = "Mes seleccionado:"
    wsDest.Range("A1").Font.Bold = True
    
    wsDest.Range("B1").NumberFormat = "@"
    wsDest.Range("B1").Value = mesUsar
    wsDest.Range("B1").Font.Bold = True
    wsDest.Range("B1").Font.Size = 12
    wsDest.Range("B1").Interior.Color = RGB(255, 240, 180)
    wsDest.Range("B1").Borders.LineStyle = xlContinuous
    wsDest.Range("B1").HorizontalAlignment = xlCenter
    
    If listaMeses <> "" Then
        With wsDest.Range("B1").Validation
            .Delete
            On Error Resume Next
            .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Formula1:=listaMeses
            On Error GoTo 0
            .IgnoreBlank = True
            .InCellDropdown = True
        End With
    End If
    
    Dim filaActual As Long: filaActual = 3
    filaActual = ArmarBloqueResumen(wsDest, filaActual, wsPxR, "FONDO 2", "Fondo 2", mesUsar)
    filaActual = filaActual + 2
    filaActual = ArmarBloqueResumen(wsDest, filaActual, wsPxR, "FONDO 3", "Fondo 3", mesUsar)
    
    wsDest.Columns("A:D").AutoFit
    wsDest.Activate
    wsDest.Range("B1").Select
    
    Application.ScreenUpdating = True

End Sub

'==================================================================
' Recalcula Resumen Total respetando el mes en B1 (sin reconstruir todo)
'==================================================================
Sub ActualizarResumenTotal()
    
    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    Dim wsPxR As Worksheet, wsDest As Worksheet, wsCuadroLoc As Worksheet
    
    On Error Resume Next
    Set wsPxR = wb.Sheets("Pesos x Retornos")
    Set wsDest = wb.Sheets("Resumen Total")
    Set wsCuadroLoc = wb.Sheets("Cuadro Locales")
    On Error GoTo 0
    
    If wsPxR Is Nothing Or wsDest Is Nothing Then
        MsgBox "Faltan hojas. Ejecuta primero GenerarResumenTotal.", vbCritical
        Exit Sub
    End If
    
    ' Validar que el mes en B1 sea válido
    Dim mesUsar As String
    mesUsar = Trim(CStr(wsDest.Range("B1").Value))
    
    If mesUsar = "" Then
        MsgBox "B1 está vacío. Selecciona un mes del menú desplegable.", vbExclamation
        Exit Sub
    End If
    
    ' Verificar que existe en Cuadro Locales
    Dim existe As Boolean: existe = False
    If Not wsCuadroLoc Is Nothing Then
        Dim ultColLoc As Long
        ultColLoc = wsCuadroLoc.Cells(1, wsCuadroLoc.Columns.Count).End(xlToLeft).Column
        Dim c As Long
        For c = 3 To ultColLoc
            If LCase(Trim(CStr(wsCuadroLoc.Cells(1, c).Value))) = LCase(mesUsar) Then
                existe = True
                Exit For
            End If
        Next c
    End If
    
    If Not existe Then
        MsgBox "El mes '" & mesUsar & "' no está en Cuadro Locales. Selecciona uno del menú desplegable.", vbExclamation
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    ' Regenerar contenido manteniendo B1
    ' Limpiar desde fila 3 hacia abajo
    Dim ultFilaDest As Long
    ultFilaDest = wsDest.Cells(wsDest.Rows.Count, 1).End(xlUp).Row
    If ultFilaDest >= 3 Then
        wsDest.Range("A3:Z" & ultFilaDest).Clear
    End If
    
    Dim filaActual As Long: filaActual = 3
    filaActual = ArmarBloqueResumen(wsDest, filaActual, wsPxR, "FONDO 2", "Fondo 2", mesUsar)
    filaActual = filaActual + 2
    filaActual = ArmarBloqueResumen(wsDest, filaActual, wsPxR, "FONDO 3", "Fondo 3", mesUsar)
    
    wsDest.Columns("A:D").AutoFit
    
    Application.ScreenUpdating = True
    
    MsgBox "Resumen Total actualizado para el mes: " & mesUsar, vbInformation
End Sub
