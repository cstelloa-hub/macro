Sub DiagnosticoRapido()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("Pesos")
    
    ' Buscar FONDO 2 SOLO en columnas A-Z y filas 1-150
    Dim filaF2 As Long: filaF2 = 0
    Dim colF2 As Long: colF2 = 0
    Dim r As Long, c As Long
    
    For r = 60 To 150
        For c = 1 To 26
            If UCase(Trim(CStr(ws.Cells(r, c).Value))) = "FONDO 2" Then
                filaF2 = r
                colF2 = c
                Exit For
            End If
        Next c
        If filaF2 > 0 Then Exit For
    Next r
    
    If filaF2 = 0 Then
        MsgBox "NO se encontró 'FONDO 2' en filas 60-150", vbCritical
        Exit Sub
    End If
    
    Dim msj As String
    msj = "FONDO 2 está en fila " & filaF2 & ", columna " & colF2 & vbCrLf & vbCrLf
    
    ' Mostrar las 3 filas siguientes, 7 columnas a la derecha
    Dim rr As Long, cc As Long
    For rr = filaF2 To filaF2 + 3
        msj = msj & "Fila " & rr & ": "
        For cc = colF2 To colF2 + 6
            Dim val As String
            val = CStr(ws.Cells(rr, cc).Value)
            If Len(val) > 12 Then val = Left(val, 12)
            msj = msj & "[" & val & "] "
        Next cc
        msj = msj & vbCrLf
    Next rr
    
    MsgBox msj, vbInformation, "Diagnóstico"
End Sub
