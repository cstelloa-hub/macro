Sub DiagnosticoEncabezadosPesos()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("Pesos")
    
    Dim ultCol As Long, ultFila As Long
    ultCol = ws.Columns.Count
    ultFila = 300
    
    ' Buscar FONDO 2
    Dim filaF2 As Long: filaF2 = 0
    Dim colF2 As Long: colF2 = 0
    Dim r As Long, c As Long
    For r = 1 To ultFila
        For c = 1 To 50
            On Error Resume Next
            Dim valBusq As String
            valBusq = UCase(Trim(CStr(ws.Cells(r, c).Value)))
            On Error GoTo 0
            If valBusq = "FONDO 2" Then
                filaF2 = r: colF2 = c: Exit For
            End If
        Next c
        If filaF2 > 0 Then Exit For
    Next r
    
    Dim msj As String
    If filaF2 = 0 Then
        MsgBox "NO se encontró 'FONDO 2'", vbCritical
        Exit Sub
    End If
    
    msj = "FONDO 2 está en celda: " & Columna(colF2) & filaF2 & vbCrLf & vbCrLf
    msj = msj & "Contenido de las filas siguientes:" & vbCrLf & vbCrLf
    
    Dim colFin As Long: colFin = colF2 + 8
    If colFin > 100 Then colFin = 100
    
    Dim rr As Long, cc As Long
    For rr = filaF2 To filaF2 + 5
        msj = msj & "Fila " & rr & ":" & vbCrLf
        For cc = colF2 To colFin
            Dim valRaw As Variant
            Dim val As String
            Dim tipoVal As String
            
            On Error Resume Next
            valRaw = ws.Cells(rr, cc).Value
            
            If Err.Number <> 0 Then
                val = "ERROR"
                tipoVal = ""
                Err.Clear
            Else
                val = CStr(valRaw)
                tipoVal = ""
                If IsDate(valRaw) Then tipoVal = " [FECHA]"
                If IsNumeric(valRaw) And Not IsDate(valRaw) And valRaw <> "" Then tipoVal = " [NUM]"
            End If
            On Error GoTo 0
            
            If Len(val) > 20 Then val = Left(val, 20) & "..."
            msj = msj & "  " & Columna(cc) & rr & " = '" & val & "'" & tipoVal & vbCrLf
        Next cc
        msj = msj & vbCrLf
    Next rr
    
    MsgBox msj, vbInformation, "Diagnóstico Encabezados"
End Sub
