Sub AplicarFormatoTodo()
    Application.ScreenUpdating = False
    
    Dim ws As Worksheet
    For Each ws In ThisWorkbook.Sheets
        AplicarFormatoInstitucional ws
    Next ws
    
    Application.ScreenUpdating = True
    MsgBox "Formato Profuturo aplicado a todas las hojas.", vbInformation
End Sub

Sub AplicarFormatoInstitucional(ws As Worksheet)
    On Error Resume Next
    
    Dim nombre As String: nombre = LCase(ws.Name)
    
    Dim rangoUsado As Range
    Set rangoUsado = ws.UsedRange
    
    If rangoUsado Is Nothing Then Exit Sub
    
    ' Fuente Arial 8 para todo el contenido
    With rangoUsado.Font
        .Name = "Arial"
        .Size = 8
    End With
    
    ' Aplicar formato específico según tipo de hoja
    Select Case nombre
        Case "resumen"
            FormatoResumenInput ws
        Case "data locales", "data extranjeros"
            FormatoDataCruda ws
        Case "cuadro locales", "cuadro extranjeros"
            FormatoCuadros ws
        Case "base 100"
            FormatoBase100 ws
        Case "pesos x retornos"
            FormatoPxR ws
        Case "resumen total"
            FormatoResumenTotal ws
        Case "top bottom"
            FormatoTopBottom ws
    End Select
    
    On Error GoTo 0
End Sub

Private Sub FormatoResumenInput(ws As Worksheet)
    ws.Range("A1").Interior.Color = RGB(212, 12, 12)
    ws.Range("A1").Font.Color = RGB(255, 255, 255)
    ws.Range("A1").Font.Bold = True
    ws.Range("D1").Interior.Color = RGB(212, 12, 12)
    ws.Range("D1").Font.Color = RGB(255, 255, 255)
    ws.Range("D1").Font.Bold = True
    ws.Range("A2:B2").Interior.Color = RGB(245, 245, 245)
    ws.Range("A2:B2").Font.Bold = True
    ws.Range("D2:E2").Interior.Color = RGB(245, 245, 245)
    ws.Range("D2:E2").Font.Bold = True
    ws.Columns("A:E").AutoFit
End Sub

Private Sub FormatoDataCruda(ws As Worksheet)
    Dim ultFila As Long
    ultFila = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    With ws.Range("A1:E1")
        .Interior.Color = RGB(212, 12, 12)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
    End With
    
    If ultFila >= 2 Then
        ws.Range("A2:E" & ultFila).Borders.LineStyle = xlContinuous
        ws.Range("A2:E" & ultFila).Borders.Color = RGB(220, 220, 220)
    End If
    
    ws.Columns("A:E").AutoFit
End Sub

Private Sub FormatoCuadros(ws As Worksheet)
    Dim ultFila As Long, ultCol As Long
    ultFila = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    ultCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    
    If ultFila < 1 Or ultCol < 1 Then Exit Sub
    
    With ws.Range(ws.Cells(1, 1), ws.Cells(1, ultCol))
        .Interior.Color = RGB(212, 12, 12)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
        .RowHeight = 22
    End With
    
    Dim r As Long
    For r = 2 To ultFila
        If r Mod 2 = 0 Then
            ws.Range(ws.Cells(r, 1), ws.Cells(r, ultCol)).Interior.Color = RGB(245, 245, 245)
        End If
    Next r
    
    ws.Range(ws.Cells(2, 1), ws.Cells(ultFila, ultCol)).Borders.LineStyle = xlContinuous
    ws.Range(ws.Cells(2, 1), ws.Cells(ultFila, ultCol)).Borders.Color = RGB(220, 220, 220)
    
    ws.Columns.AutoFit
End Sub

Private Sub FormatoBase100(ws As Worksheet)
    Dim ultFila As Long, ultCol As Long
    ultFila = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    If ultFila < 1 Then Exit Sub
    
    Dim r As Long
    For r = 1 To ultFila
        Dim val As String: val = CStr(ws.Cells(r, 1).Value)
        
        If InStr(val, "Retorno Acumulado") > 0 Then
            ultCol = ws.Cells(r, ws.Columns.Count).End(xlToLeft).Column
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, ultCol))
                .Interior.Color = RGB(212, 12, 12)
                .Font.Color = RGB(255, 255, 255)
                .Font.Bold = True
                .Font.Size = 12
                .VerticalAlignment = xlCenter
                .RowHeight = 24
            End With
            
            If r + 1 <= ultFila Then
                With ws.Range(ws.Cells(r + 1, 1), ws.Cells(r + 1, ultCol))
                    .Interior.Color = RGB(245, 245, 245)
                    .Font.Bold = True
                    .Borders.LineStyle = xlContinuous
                End With
            End If
        End If
    Next r
    
    ws.Columns.AutoFit
End Sub

Private Sub FormatoPxR(ws As Worksheet)
    Dim ultFila As Long
    ultFila = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    If ultFila < 1 Then Exit Sub
    
    Dim r As Long, ultCol As Long
    For r = 1 To ultFila
        Dim val As String: val = UCase(Trim(CStr(ws.Cells(r, 1).Value)))
        
        If val = "LOCAL" Or val = "EXTRANJERO" Then
            ws.Cells(r, 1).Font.Color = RGB(212, 12, 12)
            ws.Cells(r, 1).Font.Bold = True
            ws.Cells(r, 1).Font.Size = 14
        End If
        
        If LCase(val) = "fondo 2" Or LCase(val) = "fondo 3" Then
            ultCol = ws.Cells(r, ws.Columns.Count).End(xlToLeft).Column
            If ultCol < 5 Then ultCol = 12
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, ultCol))
                .Interior.Color = RGB(212, 12, 12)
                .Font.Color = RGB(255, 255, 255)
                .Font.Bold = True
                .Font.Size = 12
            End With
            
            If r + 1 <= ultFila Then
                ultCol = ws.Cells(r + 1, ws.Columns.Count).End(xlToLeft).Column
                With ws.Range(ws.Cells(r + 1, 1), ws.Cells(r + 1, ultCol))
                    .Interior.Color = RGB(245, 245, 245)
                    .Font.Bold = True
                End With
            End If
        End If
        
        If val = "MTD" Then
            ultCol = ws.Cells(r, ws.Columns.Count).End(xlToLeft).Column
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, ultCol))
                .Interior.Color = RGB(255, 245, 200)
                .Font.Bold = True
                .Borders.LineStyle = xlContinuous
            End With
        End If
    Next r
    
    ws.Columns.AutoFit
End Sub

Private Sub FormatoResumenTotal(ws As Worksheet)
    With ws.Range("A1")
        .Font.Bold = True
    End With
    
    With ws.Range("B1")
        .Interior.Color = RGB(255, 240, 180)
        .Font.Bold = True
        .Font.Size = 12
        .HorizontalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(180, 180, 180)
    End With
    
    Dim ultFila As Long
    ultFila = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    Dim r As Long
    For r = 3 To ultFila
        Dim val As String: val = Trim(CStr(ws.Cells(r, 1).Value))
        Dim valU As String: valU = UCase(val)
        
        If valU = "FONDO 2" Or valU = "FONDO 3" Then
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, 3))
                .Interior.Color = RGB(212, 12, 12)
                .Font.Color = RGB(255, 255, 255)
                .Font.Bold = True
                .Font.Size = 14
                .HorizontalAlignment = xlLeft
                .RowHeight = 24
                .VerticalAlignment = xlCenter
            End With
        End If
        
        If valU = "CATEGORÍA" Or valU = "CATEGORIA" Then
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, 3))
                .Interior.Color = RGB(70, 70, 70)
                .Font.Color = RGB(255, 255, 255)
                .Font.Bold = True
                .HorizontalAlignment = xlCenter
                .Borders.LineStyle = xlContinuous
            End With
        End If
        
        If valU = "LOCAL TOTAL" Or valU = "EXTRANJERO TOTAL" Then
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, 3))
                .Interior.Color = RGB(255, 245, 200)
                .Font.Bold = True
                .Borders.LineStyle = xlContinuous
            End With
        End If
        
        If InStr(valU, "TOTAL FONDO") > 0 Then
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, 3))
                .Interior.Color = RGB(212, 12, 12)
                .Font.Color = RGB(255, 255, 255)
                .Font.Bold = True
                .Font.Size = 12
                .Borders.LineStyle = xlContinuous
                .RowHeight = 22
            End With
        End If
    Next r
    
    ws.Columns("A:D").AutoFit
    If ws.Columns("A").ColumnWidth < 22 Then ws.Columns("A").ColumnWidth = 22
End Sub

Private Sub FormatoTopBottom(ws As Worksheet)
    Dim ultFila As Long
    ultFila = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    With ws.Range("A1")
        .Font.Bold = True
        .Font.Size = 16
        .Font.Color = RGB(212, 12, 12)
    End With
    
    With ws.Range("B2")
        .Interior.Color = RGB(255, 240, 180)
        .Font.Bold = True
        .Font.Size = 12
        .HorizontalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
    End With
    
    Dim r As Long
    For r = 3 To ultFila
        Dim val As String: val = Trim(CStr(ws.Cells(r, 1).Value))
        Dim valU As String: valU = UCase(val)
        
        If InStr(valU, "FONDO 2") > 0 Or InStr(valU, "FONDO 3") > 0 Then
            If InStr(valU, "LOCAL") > 0 Or InStr(valU, "EXTRANJERO") > 0 Then
                With ws.Range(ws.Cells(r, 1), ws.Cells(r, 12))
                    .Interior.Color = RGB(212, 12, 12)
                    .Font.Color = RGB(255, 255, 255)
                    .Font.Bold = True
                    .Font.Size = 13
                    .HorizontalAlignment = xlLeft
                    .VerticalAlignment = xlCenter
                    .RowHeight = 24
                End With
            End If
        End If
        
        If Left(val, 4) = "MTD " Or Left(val, 4) = "YTD " Then
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, 12))
                .Interior.Color = RGB(80, 80, 80)
                .Font.Color = RGB(255, 255, 255)
                .Font.Bold = True
                .Font.Size = 11
                .HorizontalAlignment = xlLeft
            End With
        End If
        
        If valU = "TOP (POSITIVOS)" Then
            ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Interior.Color = RGB(200, 230, 200)
            ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Font.Bold = True
        End If
        If UCase(Trim(CStr(ws.Cells(r, 7).Value))) = "BOTTOM (NEGATIVOS)" Then
            ws.Range(ws.Cells(r, 7), ws.Cells(r, 12)).Interior.Color = RGB(230, 200, 200)
            ws.Range(ws.Cells(r, 7), ws.Cells(r, 12)).Font.Bold = True
        End If
        
        If InStr(valU, "SUMA POSITIVOS") > 0 Then
            ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Interior.Color = RGB(245, 245, 245)
            ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Font.Bold = True
            ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Borders.LineStyle = xlContinuous
        End If
        If InStr(UCase(Trim(CStr(ws.Cells(r, 7).Value))), "SUMA NEGATIVOS") > 0 Then
            ws.Range(ws.Cells(r, 7), ws.Cells(r, 12)).Interior.Color = RGB(245, 245, 245)
            ws.Range(ws.Cells(r, 7), ws.Cells(r, 12)).Font.Bold = True
            ws.Range(ws.Cells(r, 7), ws.Cells(r, 12)).Borders.LineStyle = xlContinuous
        End If
        
        If valU = "NETO:" Then
            With ws.Range(ws.Cells(r, 1), ws.Cells(r, 6))
                .Interior.Color = RGB(255, 245, 200)
                .Font.Bold = True
                .Borders.LineStyle = xlContinuous
                .Borders.Weight = xlMedium
            End With
        End If
    Next r
    
    ws.Columns("A:L").AutoFit
End Sub
