Sub GenerarTopBottomContribuidores()
    GenerarTopBottomInterno True  ' True = generar gráficos
End Sub

Sub ActualizarTopBottom()
    GenerarTopBottomInterno True
    MsgBox "Top/Bottom actualizado.", vbInformation
End Sub

Private Sub GenerarTopBottomInterno(conGraficos As Boolean)
    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    Dim wsPxR As Worksheet, wsResumen As Worksheet, wsCuadroLoc As Worksheet, wsCuadroExt As Worksheet
    
    On Error Resume Next
    Set wsPxR = wb.Sheets("Pesos x Retornos")
    Set wsResumen = wb.Sheets("Resumen Total")
    Set wsCuadroLoc = wb.Sheets("Cuadro Locales")
    Set wsCuadroExt = wb.Sheets("Cuadro Extranjeros")
    On Error GoTo 0
    
    If wsPxR Is Nothing Then
        MsgBox "Falta hoja 'Pesos x Retornos'.", vbCritical
        Exit Sub
    End If
    
    Dim mesSel As String: mesSel = ""
    If Not wsResumen Is Nothing Then
        mesSel = Trim(CStr(wsResumen.Range("B1").Value))
    End If
    
    If mesSel = "" Then
        mesSel = InputBox("Escribe el mes a analizar (ej. may-26):", "Top/Bottom", "may-26")
        If mesSel = "" Then Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    Dim wsDest As Worksheet
    Set wsDest = ObtenerHoja(wb, "Top Bottom")
    
    ' Borrar gráficos previos
    Dim ch As ChartObject
    For Each ch In wsDest.ChartObjects
        ch.Delete
    Next ch
    wsDest.Cells.Clear
    
    ' Encabezado
    wsDest.Range("A1").Value = "Análisis de Contribuidores (80% Pareto)"
    wsDest.Range("A1").Font.Bold = True
    wsDest.Range("A1").Font.Size = 16
    wsDest.Range("A1").Font.Color = RGB(180, 30, 30)
    
    wsDest.Range("A2").Value = "Mes seleccionado:"
    wsDest.Range("A2").Font.Bold = True
    wsDest.Range("B2").Value = mesSel
    wsDest.Range("B2").Font.Bold = True
    wsDest.Range("B2").Font.Size = 12
    wsDest.Range("B2").Interior.Color = RGB(255, 240, 180)
    wsDest.Range("B2").HorizontalAlignment = xlCenter
    wsDest.Range("B2").Borders.LineStyle = xlContinuous
    
    wsDest.Range("D2").Value = "(Cambia el mes en Resumen Total B1 y vuelve a ejecutar)"
    wsDest.Range("D2").Font.Italic = True
    wsDest.Range("D2").Font.Color = RGB(120, 120, 120)
    
    Dim filaActual As Long: filaActual = 4
    
    filaActual = ArmarBloqueTopBottom(wsDest, filaActual, wsPxR, wsCuadroLoc, "LOCAL", "Fondo 2", "FONDO 2 - LOCAL", mesSel, conGraficos)
    filaActual = filaActual + 3
    
    filaActual = ArmarBloqueTopBottom(wsDest, filaActual, wsPxR, wsCuadroExt, "EXTRANJERO", "Fondo 2", "FONDO 2 - EXTRANJERO", mesSel, conGraficos)
    filaActual = filaActual + 3
    
    filaActual = ArmarBloqueTopBottom(wsDest, filaActual, wsPxR, wsCuadroLoc, "LOCAL", "Fondo 3", "FONDO 3 - LOCAL", mesSel, conGraficos)
    filaActual = filaActual + 3
    
    filaActual = ArmarBloqueTopBottom(wsDest, filaActual, wsPxR, wsCuadroExt, "EXTRANJERO", "Fondo 3", "FONDO 3 - EXTRANJERO", mesSel, conGraficos)
    
    wsDest.Columns("A:L").AutoFit
    wsDest.Activate
    wsDest.Range("A1").Select
    
    Application.ScreenUpdating = True
End Sub

Private Function ArmarBloqueTopBottom(wsDest As Worksheet, filaInicio As Long, _
                                       wsPxR As Worksheet, wsCuadro As Worksheet, _
                                       origenBuscar As String, fondoBuscar As String, _
                                       titulo As String, mesSel As String, _
                                       conGraficos As Boolean) As Long
    
    Dim fila As Long: fila = filaInicio
    
    ' Título del bloque
    wsDest.Cells(fila, 1).Value = titulo
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Font.Size = 14
    wsDest.Cells(fila, 1).Font.Color = RGB(255, 255, 255)
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 12)).Interior.Color = RGB(70, 110, 170)
    fila = fila + 1
    
    Dim info As String
    info = LocalizarSeccionPxR(wsPxR, origenBuscar, fondoBuscar)
    
    If info = "" Then
        wsDest.Cells(fila, 1).Value = "(no se encontró sección)"
        ArmarBloqueTopBottom = fila + 1
        Exit Function
    End If
    
    Dim partes() As String
    partes = Split(info, "|")
    Dim filaEnc As Long: filaEnc = CLng(partes(0))
    Dim colDic25 As Long: colDic25 = CLng(partes(1))
    Dim colUltMes As Long: colUltMes = CLng(partes(2))
    Dim filaMTD As Long: filaMTD = CLng(partes(3))
    
    Dim esTrimestral As Boolean: esTrimestral = False
    If colUltMes > colDic25 Then
        Dim primerEncab As String
        primerEncab = LCase(Trim(CStr(wsPxR.Cells(filaEnc, colDic25 + 1).Value)))
        If InStr(primerEncab, "q") = 1 Then esTrimestral = True
    End If
    
    Dim colMesSel As Long: colMesSel = 0
    Dim c As Long
    If esTrimestral Then
        Dim trObj As String: trObj = MesATrimestre(mesSel)
        For c = colDic25 + 1 To colUltMes
            If LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value))) = LCase(trObj) Then
                colMesSel = c
                Exit For
            End If
        Next c
        If colMesSel = 0 Then
            Dim mc As Long: mc = 0
            For c = colDic25 + 1 To colUltMes
                Dim tre As String: tre = Trim(CStr(wsPxR.Cells(filaEnc, c).Value))
                If tre <> "" Then
                    If TrimestreComparable(tre) <= TrimestreComparable(trObj) Then mc = c
                End If
            Next c
            colMesSel = mc
        End If
    Else
        For c = colDic25 To colUltMes
            If LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value))) = LCase(mesSel) Then
                colMesSel = c
                Exit For
            End If
        Next c
    End If
    
    If colMesSel = 0 Then
        wsDest.Cells(fila, 1).Value = "(no se encontró el mes " & mesSel & ")"
        ArmarBloqueTopBottom = fila + 1
        Exit Function
    End If
    
    Dim primerMesCol As Long: primerMesCol = colDic25 + 1
    
    ' Leer todos los fondos del bloque
    Dim fondos As Object: Set fondos = CreateObject("Scripting.Dictionary")
    
    Dim rr As Long
    For rr = filaEnc + 1 To filaMTD - 1
        Dim nombre As String, codigo As String, tipo As String
        nombre = Trim(CStr(wsPxR.Cells(rr, 1).Value))
        codigo = Trim(CStr(wsPxR.Cells(rr, 2).Value))
        tipo = Trim(CStr(wsPxR.Cells(rr, 3).Value))
        
        If codigo = "" Then GoTo SiguienteFila
        
        Dim mtdF As Double: mtdF = 0
        Dim vMtd As Variant: vMtd = wsPxR.Cells(rr, colMesSel).Value
        If IsNumeric(vMtd) Then mtdF = CDbl(vMtd)
        
        ' YTD = SUMA SIMPLE de contribuciones
        Dim ytdF As Double: ytdF = 0
        If colMesSel >= primerMesCol Then
            Dim cc As Long
            For cc = primerMesCol To colMesSel
                Dim vR As Variant: vR = wsPxR.Cells(rr, cc).Value
                If IsNumeric(vR) Then ytdF = ytdF + CDbl(vR)
            Next cc
        End If
        
        ' Retorno y peso aproximados desde Cuadro
        Dim retMes As Double: retMes = 0
        Dim pesoApprox As Double: pesoApprox = 0
        
        Dim filaCuadro As Long: filaCuadro = 0
        Dim ultFC As Long: ultFC = wsCuadro.Cells(wsCuadro.Rows.Count, 2).End(xlUp).Row
        Dim rc As Long
        For rc = 2 To ultFC
            If Trim(CStr(wsCuadro.Cells(rc, 2).Value)) = codigo Then
                filaCuadro = rc
                Exit For
            End If
        Next rc
        
        If filaCuadro > 0 Then
            Dim ultCC As Long: ultCC = wsCuadro.Cells(1, wsCuadro.Columns.Count).End(xlToLeft).Column
            Dim cm As Long, mesABuscar As String
            If esTrimestral Then
                mesABuscar = MesATrimestre(mesSel)
            Else
                mesABuscar = mesSel
            End If
            For cm = 3 To ultCC
                If LCase(Trim(CStr(wsCuadro.Cells(1, cm).Value))) = LCase(mesABuscar) Then
                    Dim vRet As Variant: vRet = wsCuadro.Cells(filaCuadro, cm).Value
                    If IsNumeric(vRet) Then retMes = CDbl(vRet)
                    Exit For
                End If
            Next cm
        End If
        
        If retMes <> 0 Then pesoApprox = mtdF / retMes
        
        Dim datos As String
        datos = nombre & "|" & codigo & "|" & tipo & "|" & _
                pesoApprox & "|" & retMes & "|" & mtdF & "|" & ytdF
        fondos(codigo) = datos
        
SiguienteFila:
    Next rr
    
    If fondos.Count = 0 Then
        wsDest.Cells(fila, 1).Value = "(no hay fondos en este bloque)"
        ArmarBloqueTopBottom = fila + 1
        Exit Function
    End If
    
    ' ============================
    ' SECCIÓN MTD
    ' ============================
    wsDest.Cells(fila, 1).Value = "MTD - " & mesSel
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Font.Size = 12
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 12)).Interior.Color = RGB(200, 215, 235)
    fila = fila + 1
    
    Dim filaInicioMTD As Long: filaInicioMTD = fila
    fila = EscribirParetoTopBottom(wsDest, fila, fondos, 5, 6, conGraficos)  ' índices 5=MTD, 6=YTD
    Dim filaFinMTD As Long: filaFinMTD = fila - 1
    
    fila = fila + 2
    
    ' ============================
    ' SECCIÓN YTD
    ' ============================
    wsDest.Cells(fila, 1).Value = "YTD - hasta " & mesSel
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Font.Size = 12
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 12)).Interior.Color = RGB(220, 230, 200)
    fila = fila + 1
    
    Dim filaInicioYTD As Long: filaInicioYTD = fila
    fila = EscribirParetoTopBottom(wsDest, fila, fondos, 6, 5, conGraficos)
    Dim filaFinYTD As Long: filaFinYTD = fila - 1
    
    ArmarBloqueTopBottom = fila
End Function

' Escribe el top y bottom (80% pareto) usando el índice indicado
' idxOrden = 5 (MTD) o 6 (YTD)
Private Function EscribirParetoTopBottom(wsDest As Worksheet, filaInicio As Long, _
                                          fondos As Object, idxOrden As Long, idxOtro As Long, _
                                          conGraficos As Boolean) As Long
    Dim fila As Long: fila = filaInicio
    
    ' Encabezados
    wsDest.Cells(fila, 1).Value = "TOP (positivos)"
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Interior.Color = RGB(200, 230, 200)
    wsDest.Cells(fila, 7).Value = "BOTTOM (negativos)"
    wsDest.Cells(fila, 7).Font.Bold = True
    wsDest.Cells(fila, 7).Interior.Color = RGB(230, 200, 200)
    fila = fila + 1
    
    wsDest.Cells(fila, 1).Value = "Fondo"
    wsDest.Cells(fila, 2).Value = "Tipo"
    wsDest.Cells(fila, 3).Value = "Peso"
    wsDest.Cells(fila, 4).Value = "Retorno"
    wsDest.Cells(fila, 5).Value = "Contrib."
    wsDest.Cells(fila, 6).Value = "% Acum."
    wsDest.Cells(fila, 7).Value = "Fondo"
    wsDest.Cells(fila, 8).Value = "Tipo"
    wsDest.Cells(fila, 9).Value = "Peso"
    wsDest.Cells(fila, 10).Value = "Retorno"
    wsDest.Cells(fila, 11).Value = "Contrib."
    wsDest.Cells(fila, 12).Value = "% Acum."
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 12)).Font.Bold = True
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 12)).Interior.Color = RGB(240, 240, 220)
    fila = fila + 1
    
    ' Separar positivos y negativos por el índice elegido
    Dim positivos As Object: Set positivos = CreateObject("Scripting.Dictionary")
    Dim negativos As Object: Set negativos = CreateObject("Scripting.Dictionary")
    
    Dim sumaPos As Double: sumaPos = 0
    Dim sumaNeg As Double: sumaNeg = 0
    
    Dim k As Variant
    For Each k In fondos.keys
        Dim p() As String: p = Split(CStr(fondos(k)), "|")
        Dim valOrden As Double: valOrden = CDbl(p(idxOrden))
        If valOrden > 0 Then
            positivos(k) = valOrden
            sumaPos = sumaPos + valOrden
        ElseIf valOrden < 0 Then
            negativos(k) = valOrden
            sumaNeg = sumaNeg + valOrden
        End If
    Next k
    
    Dim listaPos As Collection: Set listaPos = OrdenarDictPorValor(positivos, True)
    Dim listaNeg As Collection: Set listaNeg = OrdenarDictPorValor(negativos, False)
    
    Dim filaT As Long: filaT = fila
    Dim filaB As Long: filaB = fila
    Dim acumP As Double: acumP = 0
    Dim acumN As Double: acumN = 0
    
    Dim i As Long
    
    ' Top
    For i = 1 To listaPos.Count
        Dim codP As String: codP = CStr(listaPos(i))
        Dim datosP() As String: datosP = Split(CStr(fondos(codP)), "|")
        Dim valP As Double: valP = CDbl(datosP(idxOrden))
        Dim pctP As Double: pctP = 0
        If sumaPos > 0 Then pctP = valP / sumaPos
        acumP = acumP + pctP
        
        wsDest.Cells(filaT, 1).Value = datosP(0)
        wsDest.Cells(filaT, 2).Value = datosP(2)
        wsDest.Cells(filaT, 3).Value = CDbl(datosP(3))
        wsDest.Cells(filaT, 3).NumberFormat = "0.00%"
        wsDest.Cells(filaT, 4).Value = CDbl(datosP(4))
        wsDest.Cells(filaT, 4).NumberFormat = "0.00%"
        wsDest.Cells(filaT, 5).Value = valP
        wsDest.Cells(filaT, 5).NumberFormat = "0.0000%"
        wsDest.Cells(filaT, 6).Value = acumP
        wsDest.Cells(filaT, 6).NumberFormat = "0.0%"
        
        filaT = filaT + 1
        If acumP >= 0.8 Then Exit For
    Next i
    
    ' Bottom
    For i = 1 To listaNeg.Count
        Dim codN As String: codN = CStr(listaNeg(i))
        Dim datosN() As String: datosN = Split(CStr(fondos(codN)), "|")
        Dim valN As Double: valN = CDbl(datosN(idxOrden))
        Dim pctN As Double: pctN = 0
        If sumaNeg < 0 Then pctN = valN / sumaNeg
        acumN = acumN + pctN
        
        wsDest.Cells(filaB, 7).Value = datosN(0)
        wsDest.Cells(filaB, 8).Value = datosN(2)
        wsDest.Cells(filaB, 9).Value = CDbl(datosN(3))
        wsDest.Cells(filaB, 9).NumberFormat = "0.00%"
        wsDest.Cells(filaB, 10).Value = CDbl(datosN(4))
        wsDest.Cells(filaB, 10).NumberFormat = "0.00%"
        wsDest.Cells(filaB, 11).Value = valN
        wsDest.Cells(filaB, 11).NumberFormat = "0.0000%"
        wsDest.Cells(filaB, 12).Value = acumN
        wsDest.Cells(filaB, 12).NumberFormat = "0.0%"
        
        filaB = filaB + 1
        If acumN >= 0.8 Then Exit For
    Next i
    
    fila = Application.WorksheetFunction.Max(filaT, filaB)
    
    ' Totales
    wsDest.Cells(fila, 1).Value = "Suma positivos:"
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 5).Value = sumaPos
    wsDest.Cells(fila, 5).NumberFormat = "0.0000%"
    wsDest.Cells(fila, 5).Font.Bold = True
    
    wsDest.Cells(fila, 7).Value = "Suma negativos:"
    wsDest.Cells(fila, 7).Font.Bold = True
    wsDest.Cells(fila, 11).Value = sumaNeg
    wsDest.Cells(fila, 11).NumberFormat = "0.0000%"
    wsDest.Cells(fila, 11).Font.Bold = True
    fila = fila + 1
    
    wsDest.Cells(fila, 1).Value = "Neto:"
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 5).Value = sumaPos + sumaNeg
    wsDest.Cells(fila, 5).NumberFormat = "0.0000%"
    wsDest.Cells(fila, 5).Font.Bold = True
    wsDest.Cells(fila, 5).Interior.Color = RGB(255, 240, 180)
    fila = fila + 1
    
    ' Gráfico Pareto si se requiere
    If conGraficos Then
        CrearGraficoPareto wsDest, filaInicio, filaT - 1, filaB - 1, _
                            "Top Contribuidores", "Bottom Contribuidores"
    End If
    
    EscribirParetoTopBottom = fila
End Function

Private Sub CrearGraficoPareto(wsDest As Worksheet, filaEncab As Long, _
                                ultFilaTop As Long, ultFilaBot As Long, _
                                tituloTop As String, tituloBot As String)
    
    Dim filaDatosInicio As Long: filaDatosInicio = filaEncab + 2
    
    ' Gráfico Top
    If ultFilaTop >= filaDatosInicio Then
        Dim leftPos1 As Double, topPos1 As Double
        leftPos1 = wsDest.Cells(filaEncab, 14).Left
        topPos1 = wsDest.Cells(filaEncab, 14).Top
        
        Dim ch1 As ChartObject
        Set ch1 = wsDest.ChartObjects.Add(Left:=leftPos1, Top:=topPos1, Width:=380, Height:=220)
        
        With ch1.Chart
            .ChartType = xlColumnClustered
            
            ' Limpiar series existentes
            Do While .SeriesCollection.Count > 0
                .SeriesCollection(1).Delete
            Loop
            
            ' Serie de contribuciones (barras)
            Dim sBarra As Series
            Set sBarra = .SeriesCollection.NewSeries
            sBarra.Name = "Contribución"
            sBarra.Values = wsDest.Range(wsDest.Cells(filaDatosInicio, 5), wsDest.Cells(ultFilaTop, 5))
            sBarra.XValues = wsDest.Range(wsDest.Cells(filaDatosInicio, 1), wsDest.Cells(ultFilaTop, 1))
            On Error Resume Next
            sBarra.Format.Fill.ForeColor.RGB = RGB(70, 130, 180)
            On Error GoTo 0
            
            ' Serie acumulado (línea) en eje secundario
            Dim sLinea As Series
            Set sLinea = .SeriesCollection.NewSeries
            sLinea.Name = "% Acumulado"
            sLinea.Values = wsDest.Range(wsDest.Cells(filaDatosInicio, 6), wsDest.Cells(ultFilaTop, 6))
            sLinea.ChartType = xlLineMarkers
            On Error Resume Next
            sLinea.AxisGroup = xlSecondary
            sLinea.Format.Line.ForeColor.RGB = RGB(220, 50, 50)
            sLinea.Format.Line.Weight = 2.25
            On Error GoTo 0
            
            .HasTitle = True
            .ChartTitle.Text = tituloTop
            .ChartTitle.Font.Size = 11
            .ChartTitle.Font.Bold = True
            
            .HasLegend = True
            .Legend.Position = xlLegendPositionBottom
            .Legend.Font.Size = 8
            
            On Error Resume Next
            .Axes(xlValue, xlPrimary).TickLabels.NumberFormat = "0.0000%"
            .Axes(xlValue, xlPrimary).TickLabels.Font.Size = 8
            .Axes(xlValue, xlSecondary).TickLabels.NumberFormat = "0%"
            .Axes(xlValue, xlSecondary).TickLabels.Font.Size = 8
            .Axes(xlValue, xlSecondary).MaximumScale = 1
            .Axes(xlValue, xlSecondary).MinimumScale = 0
            .Axes(xlCategory).TickLabels.Font.Size = 7
            .Axes(xlCategory).TickLabelPosition = xlLow
            On Error GoTo 0
        End With
    End If
    
    ' Gráfico Bottom
    If ultFilaBot >= filaDatosInicio Then
        Dim leftPos2 As Double, topPos2 As Double
        leftPos2 = wsDest.Cells(filaEncab, 14).Left
        topPos2 = wsDest.Cells(filaEncab + 12, 14).Top
        
        Dim ch2 As ChartObject
        Set ch2 = wsDest.ChartObjects.Add(Left:=leftPos2, Top:=topPos2, Width:=380, Height:=220)
        
        With ch2.Chart
            .ChartType = xlColumnClustered
            
            Do While .SeriesCollection.Count > 0
                .SeriesCollection(1).Delete
            Loop
            
            Dim sBarra2 As Series
            Set sBarra2 = .SeriesCollection.NewSeries
            sBarra2.Name = "Contribución"
            sBarra2.Values = wsDest.Range(wsDest.Cells(filaDatosInicio, 11), wsDest.Cells(ultFilaBot, 11))
            sBarra2.XValues = wsDest.Range(wsDest.Cells(filaDatosInicio, 7), wsDest.Cells(ultFilaBot, 7))
            On Error Resume Next
            sBarra2.Format.Fill.ForeColor.RGB = RGB(220, 100, 100)
            On Error GoTo 0
            
            Dim sLinea2 As Series
            Set sLinea2 = .SeriesCollection.NewSeries
            sLinea2.Name = "% Acumulado"
            sLinea2.Values = wsDest.Range(wsDest.Cells(filaDatosInicio, 12), wsDest.Cells(ultFilaBot, 12))
            sLinea2.ChartType = xlLineMarkers
            On Error Resume Next
            sLinea2.AxisGroup = xlSecondary
            sLinea2.Format.Line.ForeColor.RGB = RGB(50, 50, 50)
            sLinea2.Format.Line.Weight = 2.25
            On Error GoTo 0
            
            .HasTitle = True
            .ChartTitle.Text = tituloBot
            .ChartTitle.Font.Size = 11
            .ChartTitle.Font.Bold = True
            
            .HasLegend = True
            .Legend.Position = xlLegendPositionBottom
            .Legend.Font.Size = 8
            
            On Error Resume Next
            .Axes(xlValue, xlPrimary).TickLabels.NumberFormat = "0.0000%"
            .Axes(xlValue, xlPrimary).TickLabels.Font.Size = 8

            .Axes(xlValue, xlSecondary).TickLabels.Font.Size = 8

            .Axes(xlCategory).TickLabels.Font.Size = 7

            On Error GoTo 0
        End With
    End If
    
End Sub

Private Function OrdenarDictPorValor(dic As Object, desc As Boolean) As Collection
    Dim resultado As Collection
    Set resultado = New Collection
    
    If dic.Count = 0 Then
        Set OrdenarDictPorValor = resultado
        Exit Function
    End If
    
    Dim arr() As Variant
    ReDim arr(1 To dic.Count, 1 To 2)
    
    Dim i As Long: i = 0
    Dim k As Variant
    For Each k In dic.keys
        i = i + 1
        arr(i, 1) = CStr(k)
        arr(i, 2) = CDbl(dic(k))
    Next k
    
    Dim j As Long
    Dim tmpK As Variant, tmpV As Variant
    For i = 1 To dic.Count - 1
        For j = i + 1 To dic.Count
            Dim cambiar As Boolean: cambiar = False
            If desc Then
                If arr(i, 2) < arr(j, 2) Then cambiar = True
            Else
                If arr(i, 2) > arr(j, 2) Then cambiar = True
            End If
            
            If cambiar Then
                tmpK = arr(i, 1): tmpV = arr(i, 2)
                arr(i, 1) = arr(j, 1): arr(i, 2) = arr(j, 2)
                arr(j, 1) = tmpK: arr(j, 2) = tmpV
            End If
        Next j
    Next i
    
    For i = 1 To dic.Count
        resultado.Add CStr(arr(i, 1))
    Next i
    
    Set OrdenarDictPorValor = resultado
End Function

vbaSub ActualizarResumenTotal()
    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    Dim wsPxR As Worksheet, wsDest As Worksheet, wsCuadroLoc As Worksheet
    
    On Error Resume Next
    Set wsPxR = wb.Sheets("Pesos x Retornos")
    Set wsDest = wb.Sheets("Resumen Total")
    Set wsCuadroLoc = wb.Sheets("Cuadro Locales")
    On Error GoTo 0
    
    If wsPxR Is Nothing Or wsDest Is Nothing Then
        MsgBox "Faltan hojas. Ejecuta primero GenerarResumenTotal.", vbCritical
        Exit Sub
    End If
    
    Dim mesUsar As String
    mesUsar = Trim(CStr(wsDest.Range("B1").Value))
    
    If mesUsar = "" Then
        MsgBox "B1 está vacío. Selecciona un mes del menú desplegable.", vbExclamation
        Exit Sub
    End If
    
    Dim existe As Boolean: existe = False
    If Not wsCuadroLoc Is Nothing Then
        Dim ultColLoc As Long
        ultColLoc = wsCuadroLoc.Cells(1, wsCuadroLoc.Columns.Count).End(xlToLeft).Column
        Dim c As Long
        For c = 3 To ultColLoc
            If LCase(Trim(CStr(wsCuadroLoc.Cells(1, c).Value))) = LCase(mesUsar) Then
                existe = True
                Exit For
            End If
        Next c
    End If
    
    If Not existe Then
        MsgBox "El mes '" & mesUsar & "' no está en Cuadro Locales. Selecciona uno del menú desplegable.", vbExclamation
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    Dim ultFilaDest As Long
    ultFilaDest = wsDest.Cells(wsDest.Rows.Count, 1).End(xlUp).Row
    If ultFilaDest >= 3 Then
        wsDest.Range("A3:Z" & ultFilaDest).Clear
    End If
    
    Dim filaActual As Long: filaActual = 3
    filaActual = ArmarBloqueResumen(wsDest, filaActual, wsPxR, "FONDO 2", "Fondo 2", mesUsar)
    filaActual = filaActual + 2
    filaActual = ArmarBloqueResumen(wsDest, filaActual, wsPxR, "FONDO 3", "Fondo 3", mesUsar)
    
    wsDest.Columns("A:D").AutoFit
    
    ' ===== NUEVO: Aplicar formato Profuturo =====
    AplicarFormatoInstitucional wsDest
    ' ============================================
    
    Application.ScreenUpdating = True
    
    MsgBox "Resumen Total actualizado para el mes: " & mesUsar, vbInformation
End Sub
