Private Sub VolcarData(wsMarcas As Worksheet, wsData As Worksheet)
    
    wsData.Cells.Clear
    
    ' 5 COLUMNAS (con EEFF)
    wsData.Range("A1").Value = "Código SBS"
    wsData.Range("B1").Value = "Nombre Fondo"
    wsData.Range("C1").Value = "Día"
    wsData.Range("D1").Value = "EEFF"
    wsData.Range("E1").Value = "Var.Adj."
    wsData.Range("A1:E1").Font.Bold = True
    wsData.Range("A1:E1").Interior.Color = RGB(200, 215, 235)
    
    Dim ultCol As Long, ultFila As Long
    ultCol = wsMarcas.Cells(3, wsMarcas.Columns.Count).End(xlToLeft).Column
    If ultCol < 1 Then Exit Sub
    If ultCol > 2000 Then ultCol = 2000
    
    ultFila = wsMarcas.UsedRange.Rows.Count
    If ultFila > 1000 Then ultFila = 1000
    If ultFila < 5 Then ultFila = 5
    
    Dim arrMarcas As Variant
    arrMarcas = wsMarcas.Range(wsMarcas.Cells(1, 1), wsMarcas.Cells(ultFila, ultCol)).Value
    
    Dim colsInicio As Collection
    Set colsInicio = New Collection
    
    Dim j As Long
    For j = 1 To ultCol
        Dim h As String
        h = LCase(Trim(CStr(arrMarcas(3, j))))
        If h = "día" Or h = "dia" Then colsInicio.Add j
    Next j
    
    If colsInicio.Count = 0 Then Exit Sub
    
    Dim arrSalida() As Variant
    ReDim arrSalida(1 To 100000, 1 To 5)
    Dim filaSal As Long: filaSal = 0
    
    Dim b As Long
    For b = 1 To colsInicio.Count
        Dim colIni As Long, colFin As Long
        colIni = colsInicio(b)
        If b < colsInicio.Count Then
            colFin = colsInicio(b + 1) - 1
        Else
            colFin = ultCol
        End If
        
        Dim nombre As String, codigo As String
        nombre = Trim(CStr(arrMarcas(1, colIni)))
        codigo = Trim(CStr(arrMarcas(2, colIni)))
        
        Dim colVar As Long: colVar = 0
        Dim colEEFF As Long: colEEFF = 0
        For j = colIni To colFin
            Dim hh As String
            hh = LCase(Replace(Trim(CStr(arrMarcas(3, j))), " ", ""))
            Select Case hh
                Case "var.adj.", "var.adj", "varadj", "varadj.", "variaciónajustada", "variacionajustada"
                    colVar = j
                Case "eeff", "ee.ff.", "ee.ff", "ef.fi.", "eff", "estadosfinancieros"
                    colEEFF = j
            End Select
        Next j
        
        If codigo = "" Or colVar = 0 Then GoTo SiguienteBloque
        
        Dim rr As Long, vaciosSeguidos As Long
        vaciosSeguidos = 0
        rr = 4
        Do
            If rr > ultFila Then Exit Do
            
            Dim vFch As Variant
            vFch = arrMarcas(rr, colIni)
            
            If IsDate(vFch) Then
                vaciosSeguidos = 0
                Dim vVar As Variant
                vVar = arrMarcas(rr, colVar)
                
                ' Limpiar el valor: si es texto con %, sacarle el % y dividir entre 100
                Dim vVarStr As String
                vVarStr = CStr(vVar)
                Dim teniaPorcentaje As Boolean
                teniaPorcentaje = (InStr(vVarStr, "%") > 0)
                vVarStr = Replace(vVarStr, "%", "")
                vVarStr = Replace(vVarStr, ",", ".")
                vVarStr = Trim(vVarStr)
                
                If IsNumeric(vVarStr) Then
                    Dim varNumLimpio As Double
                    varNumLimpio = CDbl(vVarStr)
                    
                    ' Si tenía %, dividir entre 100 para convertir a decimal
                    If teniaPorcentaje Then
                        varNumLimpio = varNumLimpio / 100
                    End If
                    
                    filaSal = filaSal + 1
                    arrSalida(filaSal, 1) = codigo
                    arrSalida(filaSal, 2) = nombre
                    arrSalida(filaSal, 3) = CDate(vFch)
                    
                    If colEEFF > 0 Then
                        arrSalida(filaSal, 4) = arrMarcas(rr, colEEFF)
                    Else
                        arrSalida(filaSal, 4) = ""
                    End If
                    
                    arrSalida(filaSal, 5) = varNumLimpio
                End If
            Else
                Dim txt As String
                txt = LCase(Trim(CStr(vFch)))
                If txt = "f1" Or txt = "cc" Or txt = "dd" Or txt = "nav" Or txt = "%" Then Exit Do
                If txt = "" Then
                    vaciosSeguidos = vaciosSeguidos + 1
                    If vaciosSeguidos > 30 Then Exit Do
                Else
                    Exit Do
                End If
            End If
            rr = rr + 1
        Loop
        
SiguienteBloque:
    Next b
    
    If filaSal > 0 Then
        Dim arrFinal() As Variant
        ReDim arrFinal(1 To filaSal, 1 To 5)
        Dim i As Long, kk As Long
        For i = 1 To filaSal
            For kk = 1 To 5
                arrFinal(i, kk) = arrSalida(i, kk)
            Next kk
        Next i
        
        wsData.Range("A2").Resize(filaSal, 5).Value = arrFinal
        wsData.Range("C2:C" & filaSal + 1).NumberFormat = "dd/mm/yyyy"
        wsData.Range("E2:E" & filaSal + 1).NumberFormat = "0.00%"
    End If
    
    wsData.Columns("A:E").AutoFit
End Sub


Sub VerTipoCeldaMarcas()
    Dim codBuscar As String
    codBuscar = InputBox("Código SBS del fondo:", "Verificar Marcas")
    If codBuscar = "" Then Exit Sub
    codBuscar = Trim(codBuscar)
    
    Dim rutaMarcas As Variant
    rutaMarcas = Application.GetOpenFilename( _
        FileFilter:="Archivos Excel (*.xlsx;*.xlsm), *.xlsx;*.xlsm", _
        Title:="Selecciona Marcas.xlsx")
    If rutaMarcas = False Then Exit Sub
    
    Application.ScreenUpdating = False
    Dim wbMarcas As Workbook
    Set wbMarcas = Workbooks.Open(Filename:=CStr(rutaMarcas), ReadOnly:=True, UpdateLinks:=0)
    
    Dim ws As Worksheet
    Dim hojaUsar As String
    hojaUsar = InputBox("¿Local o Extranjero? (L o E)", "Hoja")
    On Error Resume Next
    If UCase(hojaUsar) = "L" Then
        Set ws = wbMarcas.Sheets("1")
    Else
        Set ws = wbMarcas.Sheets("2")
    End If
    On Error GoTo 0
    
    Dim ultCol As Long
    ultCol = ws.Cells(3, ws.Columns.Count).End(xlToLeft).Column
    If ultCol > 2000 Then ultCol = 2000
    
    ' Buscar fondo
    Dim colFondo As Long: colFondo = 0
    Dim j As Long
    For j = 1 To ultCol
        If Trim(CStr(ws.Cells(2, j).Value)) = codBuscar Then
            colFondo = j
            Exit For
        End If
    Next j
    
    If colFondo = 0 Then
        wbMarcas.Close False
        MsgBox "No encontré el código"
        Exit Sub
    End If
    
    ' Buscar columna Var.Adj. en el bloque
    Dim colFin As Long: colFin = ultCol
    For j = colFondo + 1 To ultCol
        If Trim(CStr(ws.Cells(2, j).Value)) <> "" Then
            colFin = j - 1
            Exit For
        End If
    Next j
    
    Dim colVar As Long: colVar = 0
    For j = colFondo To colFin
        Dim hh As String
        hh = LCase(Replace(Trim(CStr(ws.Cells(3, j).Value)), " ", ""))
        If hh = "var.adj." Or hh = "var.adj" Or hh = "varadj" Or hh = "varadj." Then colVar = j
    Next j
    
    Dim msj As String
    msj = "Fondo en col " & colFondo & ", Var.Adj. en col " & colVar & vbCrLf & vbCrLf
    
    If colVar > 0 Then
        msj = msj & "PRIMERAS 3 VAR.ADJ. del fondo:" & vbCrLf & vbCrLf
        
        Dim r As Long, contador As Long: contador = 0
        For r = 4 To 100
            Dim v As Variant
            v = ws.Cells(r, colVar).Value
            
            Dim vTxt As Variant
            vTxt = ws.Cells(r, colVar).Text  ' lo que se ve en la celda
            
            If v <> "" Then
                contador = contador + 1
                msj = msj & "Fila " & r & ":" & vbCrLf
                msj = msj & "  .Value = '" & v & "'" & vbCrLf
                msj = msj & "  TypeName = " & TypeName(v) & vbCrLf
                msj = msj & "  IsNumeric = " & IsNumeric(v) & vbCrLf
                msj = msj & "  .Text (lo visible) = '" & vTxt & "'" & vbCrLf
                msj = msj & "  NumberFormat = '" & ws.Cells(r, colVar).NumberFormat & "'" & vbCrLf
                msj = msj & vbCrLf
                If contador >= 3 Then Exit For
            End If
        Next r
    End If
    
    wbMarcas.Close False
    Application.ScreenUpdating = True
    
    MsgBox msj, vbInformation
End Sub

Sub VerificarCalculoJunio()
    Dim codBuscar As String
    codBuscar = "5111102FIBR3"
    
    Dim wsCuadroLoc As Worksheet, wsPesos As Worksheet, wsPxR As Worksheet
    Set wsCuadroLoc = ThisWorkbook.Sheets("Cuadro Locales")
    Set wsPesos = ThisWorkbook.Sheets("Pesos")
    Set wsPxR = ThisWorkbook.Sheets("Pesos x Retornos")
    
    Dim msj As String
    msj = "VERIFICACIÓN PARA " & codBuscar & " - JUN-26" & vbCrLf & vbCrLf
    
    ' 1. Retorno crudo (Value) en Cuadro Locales
    Dim filaC As Long: filaC = 0
    Dim ultF As Long: ultF = wsCuadroLoc.Cells(wsCuadroLoc.Rows.Count, 2).End(xlUp).Row
    Dim r As Long
    For r = 2 To ultF
        If Trim(CStr(wsCuadroLoc.Cells(r, 2).Value)) = codBuscar Then
            filaC = r
            Exit For
        End If
    Next r
    
    If filaC = 0 Then
        MsgBox "No encontré el fondo en Cuadro Locales"
        Exit Sub
    End If
    
    Dim colJun As Long: colJun = 0
    Dim ultC As Long: ultC = wsCuadroLoc.Cells(1, wsCuadroLoc.Columns.Count).End(xlToLeft).Column
    Dim c As Long
    For c = 3 To ultC
        If LCase(Trim(CStr(wsCuadroLoc.Cells(1, c).Value))) = "jun-26" Then
            colJun = c
            Exit For
        End If
    Next c
    
    If colJun = 0 Then
        MsgBox "No encontré jun-26 en Cuadro Locales"
        Exit Sub
    End If
    
    Dim vRet As Variant
    vRet = wsCuadroLoc.Cells(filaC, colJun).Value
    
    msj = msj & "1. RETORNO en Cuadro Locales (jun-26):" & vbCrLf
    msj = msj & "   .Value = " & vRet & vbCrLf
    msj = msj & "   TypeName = " & TypeName(vRet) & vbCrLf
    msj = msj & "   Si lo multiplicas por 100: " & (CDbl(vRet) * 100) & "%" & vbCrLf & vbCrLf
    
    ' 2. Peso en hoja Pesos (versión vieja: dic-25)
    Dim pesosF2 As Object: Set pesosF2 = LeerPesosDic25(wsPesos, "FONDO 2")
    Dim pesosF3 As Object: Set pesosF3 = LeerPesosDic25(wsPesos, "FONDO 3")
    
    Dim pesoUsar As Double: pesoUsar = 0
    Dim fondoUsar As String: fondoUsar = ""
    If pesosF2.exists(codBuscar) Then
        pesoUsar = CDbl(pesosF2(codBuscar))
        fondoUsar = "F2"
    ElseIf pesosF3.exists(codBuscar) Then
        pesoUsar = CDbl(pesosF3(codBuscar))
        fondoUsar = "F3"
    End If
    
    msj = msj & "2. PESO leído (Dic-25) " & fondoUsar & ":" & vbCrLf
    msj = msj & "   Valor crudo = " & pesoUsar & vbCrLf
    msj = msj & "   En %: " & (pesoUsar * 100) & "%" & vbCrLf & vbCrLf
    
    ' 3. Cálculo manual
    Dim calc As Double
    calc = CDbl(vRet) * pesoUsar
    msj = msj & "3. CÁLCULO ESPERADO (retorno × peso):" & vbCrLf
    msj = msj & "   = " & vRet & " × " & pesoUsar & vbCrLf
    msj = msj & "   = " & calc & vbCrLf
    msj = msj & "   En %: " & (calc * 100) & "%" & vbCrLf & vbCrLf
    
    ' 4. Valor real en PxR
    Dim filaP As Long: filaP = 0
    Dim ultP As Long: ultP = wsPxR.Cells(wsPxR.Rows.Count, 2).End(xlUp).Row
    For r = 1 To ultP
        If Trim(CStr(wsPxR.Cells(r, 2).Value)) = codBuscar Then
            filaP = r
            Exit For
        End If
    Next r
    
    If filaP > 0 Then
        Dim filaEnc As Long: filaEnc = 0
        For r = filaP - 1 To 1 Step -1
            If LCase(Trim(CStr(wsPxR.Cells(r, 5).Value))) = "dic-25" Then
                filaEnc = r
                Exit For
            End If
        Next r
        
        Dim colJunP As Long: colJunP = 0
        Dim ultCP As Long: ultCP = wsPxR.Cells(filaEnc, wsPxR.Columns.Count).End(xlToLeft).Column
        For c = 5 To ultCP
            If LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value))) = "jun-26" Then
                colJunP = c
                Exit For
            End If
        Next c
        
        If colJunP > 0 Then
            Dim vPxR As Variant
            vPxR = wsPxR.Cells(filaP, colJunP).Value
            msj = msj & "4. VALOR ACTUAL en Pesos x Retornos (jun-26):" & vbCrLf
            msj = msj & "   .Value = " & vPxR & vbCrLf
            msj = msj & "   En %: " & (CDbl(vPxR) * 100) & "%" & vbCrLf
        End If
    End If
    
    MsgBox msj, vbInformation, "Verificación"
End Sub
