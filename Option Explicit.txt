Sub ActualizarTodo()
    
    Dim respuesta As VbMsgBoxResult
    respuesta = MsgBox("Esto va a:" & vbCrLf & vbCrLf & _
                       "1. Extraer data cruda de Marcas.xlsx" & vbCrLf & _
                       "2. Regenerar Cuadro Locales y Cuadro Extranjeros" & vbCrLf & _
                       "3. Regenerar Base 100 y Gráficos" & vbCrLf & _
                       "4. Regenerar Pesos x Retornos" & vbCrLf & _
                       "5. Regenerar Resumen Total" & vbCrLf & vbCrLf & _
                       "¿Continuar?", vbYesNo + vbQuestion, "Actualizar Todo")
    
    If respuesta = vbNo Then Exit Sub
    
    Dim tInicio As Double
    tInicio = Timer
    
    Application.StatusBar = "Paso 1 de 5: Extrayendo data..."
    ExtraerData
    
    Application.StatusBar = "Paso 2 de 5: Generando cuadros..."
    GenerarCuadros
    
    Application.StatusBar = "Paso 3 de 5: Generando Base 100 y Gráficos..."
    GenerarBase100YGraficos
    
    Application.StatusBar = "Paso 4 de 5: Generando Pesos x Retornos..."
    GenerarPesosXRetornos
    
    Application.StatusBar = "Paso 5 de 5: Generando Resumen Total..."
    GenerarResumenTotal
    
    Application.StatusBar = False
    
    Dim duracion As Double
    duracion = Timer - tInicio
    
    MsgBox "Actualización completa en " & Format(duracion, "0.0") & " segundos.", vbInformation, "Listo"

End Sub

'==================================================================
' MACRO 1: Extraer data cruda (OPTIMIZADA con arrays)
'==================================================================
Sub ExtraerData()

    Dim wbDestino As Workbook, wbMarcas As Workbook
    Dim wsMarcasLoc As Worksheet, wsMarcasExt As Worksheet
    Dim wsDataLoc As Worksheet, wsDataExt As Worksheet
    Dim rutaMarcas As Variant
    
    Set wbDestino = ThisWorkbook
    
    rutaMarcas = Application.GetOpenFilename( _
        FileFilter:="Archivos Excel (*.xlsx;*.xlsm), *.xlsx;*.xlsm", _
        Title:="Selecciona Marcas.xlsx")
    
    If rutaMarcas = False Then
        MsgBox "No seleccionaste ningún archivo.", vbExclamation
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    
    Set wbMarcas = Workbooks.Open(Filename:=CStr(rutaMarcas), ReadOnly:=True, UpdateLinks:=0)
    
    On Error Resume Next
    Set wsMarcasLoc = wbMarcas.Sheets("1")
    Set wsMarcasExt = wbMarcas.Sheets("2")
    On Error GoTo 0
    
    If wsMarcasLoc Is Nothing Or wsMarcasExt Is Nothing Then
        wbMarcas.Close SaveChanges:=False
        Application.ScreenUpdating = True
        Application.Calculation = xlCalculationAutomatic
        Application.EnableEvents = True
        Application.DisplayAlerts = True
        MsgBox "No encontré las hojas '1' y '2' en Marcas.xlsx.", vbCritical
        Exit Sub
    End If
    
    Set wsDataLoc = ObtenerHoja(wbDestino, "Data Locales")
    Set wsDataExt = ObtenerHoja(wbDestino, "Data Extranjeros")
    
    VolcarData wsMarcasLoc, wsDataLoc
    VolcarData wsMarcasExt, wsDataExt
    
    wbMarcas.Close SaveChanges:=False
    
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.DisplayAlerts = True

End Sub

Private Sub VolcarData(wsMarcas As Worksheet, wsData As Worksheet)
    
    wsData.Cells.Clear
    
    wsData.Range("A1").Value = "Código SBS"
    wsData.Range("B1").Value = "Nombre Fondo"
    wsData.Range("C1").Value = "Día"
    wsData.Range("D1").Value = "Var.Adj."
    wsData.Range("A1:D1").Font.Bold = True
    wsData.Range("A1:D1").Interior.Color = RGB(200, 215, 235)
    
    Dim ultCol As Long, ultFila As Long
    ultCol = wsMarcas.Cells(3, wsMarcas.Columns.Count).End(xlToLeft).Column
    If ultCol < 1 Then Exit Sub
    If ultCol > 500 Then ultCol = 500
    
    ultFila = wsMarcas.UsedRange.Rows.Count
    If ultFila > 1000 Then ultFila = 1000
    If ultFila < 5 Then ultFila = 5
    
    Dim arrMarcas As Variant
    arrMarcas = wsMarcas.Range(wsMarcas.Cells(1, 1), wsMarcas.Cells(ultFila, ultCol)).Value
    
    Dim colsInicio As Collection
    Set colsInicio = New Collection
    
    Dim j As Long
    For j = 1 To ultCol
        Dim h As String
        h = LCase(Trim(CStr(arrMarcas(3, j))))
        If h = "día" Or h = "dia" Then colsInicio.Add j
    Next j
    
    If colsInicio.Count = 0 Then Exit Sub
    
    Dim arrSalida() As Variant
    ReDim arrSalida(1 To 50000, 1 To 4)
    Dim filaSal As Long: filaSal = 0
    
    Dim b As Long
    For b = 1 To colsInicio.Count
        Dim colIni As Long, colFin As Long
        colIni = colsInicio(b)
        If b < colsInicio.Count Then
            colFin = colsInicio(b + 1) - 1
        Else
            colFin = ultCol
        End If
        
        Dim nombre As String, codigo As String
        nombre = Trim(CStr(arrMarcas(1, colIni)))
        codigo = Trim(CStr(arrMarcas(2, colIni)))
        
        Dim colVar As Long: colVar = 0
        For j = colIni To colFin
            Dim hh As String
            hh = LCase(Replace(Trim(CStr(arrMarcas(3, j))), " ", ""))
            Select Case hh
                Case "var.adj.", "var.adj", "varadj", "varadj.", "variaciónajustada", "variacionajustada"
                    colVar = j
            End Select
        Next j
        
        If codigo = "" Or colVar = 0 Then GoTo SiguienteBloque
        
        Dim rr As Long, vaciosSeguidos As Long
        vaciosSeguidos = 0
        rr = 4
        Do
            If rr > ultFila Then Exit Do
            
            Dim vFch As Variant
            vFch = arrMarcas(rr, colIni)
            
            If IsDate(vFch) Then
                vaciosSeguidos = 0
                Dim vVar As Variant
                vVar = arrMarcas(rr, colVar)
                If IsNumeric(vVar) Then
                    filaSal = filaSal + 1
                    arrSalida(filaSal, 1) = codigo
                    arrSalida(filaSal, 2) = nombre
                    arrSalida(filaSal, 3) = CDate(vFch)
                    arrSalida(filaSal, 4) = CDbl(vVar)
                End If
            Else
                Dim txt As String
                txt = LCase(Trim(CStr(vFch)))
                If txt = "f1" Or txt = "cc" Or txt = "dd" Or txt = "nav" Or txt = "%" Then Exit Do
                If txt = "" Then
                    vaciosSeguidos = vaciosSeguidos + 1
                    If vaciosSeguidos > 30 Then Exit Do
                Else
                    Exit Do
                End If
            End If
            rr = rr + 1
        Loop
        
SiguienteBloque:
    Next b
    
    If filaSal > 0 Then
        Dim arrFinal() As Variant
        ReDim arrFinal(1 To filaSal, 1 To 4)
        Dim i As Long, kk As Long
        For i = 1 To filaSal
            For kk = 1 To 4
                arrFinal(i, kk) = arrSalida(i, kk)
            Next kk
        Next i
        
        wsData.Range("A2").Resize(filaSal, 4).Value = arrFinal
        wsData.Range("C2:C" & filaSal + 1).NumberFormat = "dd/mm/yyyy"
        wsData.Range("D2:D" & filaSal + 1).NumberFormat = "0.00%"
    End If
    
    wsData.Columns("A:D").AutoFit
End Sub

'==================================================================
' MACRO 2: Generar cuadros consolidados
'==================================================================
Sub GenerarCuadros()

    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    Dim wsResumen As Worksheet, wsDataLoc As Worksheet, wsDataExt As Worksheet
    Dim wsCuadroLoc As Worksheet, wsCuadroExt As Worksheet
    
    On Error Resume Next
    Set wsResumen = wb.Sheets("Resumen")
    Set wsDataLoc = wb.Sheets("Data Locales")
    Set wsDataExt = wb.Sheets("Data Extranjeros")
    On Error GoTo 0
    
    If wsResumen Is Nothing Then
        Set wsResumen = ObtenerHoja(wb, "Resumen")
        ArmarPlantillaResumen wsResumen
        MsgBox "Se creó la plantilla 'Resumen'. Llena los datos y vuelve a ejecutar.", vbInformation
        Exit Sub
    End If
    
    If wsDataLoc Is Nothing Or wsDataExt Is Nothing Then
        MsgBox "No encontré las hojas de data cruda. Ejecuta primero 'ExtraerData'.", vbCritical
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    Set wsCuadroLoc = ObtenerHoja(wb, "Cuadro Locales")
    Set wsCuadroExt = ObtenerHoja(wb, "Cuadro Extranjeros")
    
    ArmarCuadro wsResumen, wsDataLoc, wsCuadroLoc, "LOCAL"
    ArmarCuadro wsResumen, wsDataExt, wsCuadroExt, "EXTRANJERO"
    
    Application.ScreenUpdating = True

End Sub

Private Sub ArmarCuadro(wsResumen As Worksheet, wsData As Worksheet, _
                        wsCuadro As Worksheet, tipo As String)
    
    Dim colNombre As Long, colCodigo As Long, filaInicio As Long
    If tipo = "LOCAL" Then
        colNombre = 1: colCodigo = 2: filaInicio = 3
    Else
        colNombre = 4: colCodigo = 5: filaInicio = 3
    End If
    
    Dim fondosPedidos As Object
    Set fondosPedidos = CreateObject("Scripting.Dictionary")
    Dim ordenPedidos As Collection
    Set ordenPedidos = New Collection
    
    Dim ultFila As Long, i As Long
    ultFila = wsResumen.Cells(wsResumen.Rows.Count, colCodigo).End(xlUp).Row
    
    For i = filaInicio To ultFila
        Dim cod As String, nom As String
        cod = Trim(CStr(wsResumen.Cells(i, colCodigo).Value))
        nom = Trim(CStr(wsResumen.Cells(i, colNombre).Value))
        If cod <> "" Then
            fondosPedidos(cod) = nom
            ordenPedidos.Add cod
        End If
    Next i
    
    wsCuadro.Cells.Clear
    
    If fondosPedidos.Count = 0 Then
        wsCuadro.Range("A1").Value = "No hay fondos cargados en Resumen para " & tipo
        Exit Sub
    End If
    
    Dim ultData As Long
    ultData = wsData.Cells(wsData.Rows.Count, 1).End(xlUp).Row
    
    If ultData < 2 Then
        wsCuadro.Range("A1").Value = "La hoja " & wsData.Name & " está vacía."
        Exit Sub
    End If
    
    Dim arr As Variant
    arr = wsData.Range("A2:D" & ultData).Value
    
    Dim datos As Object, periodos As Object
    Set datos = CreateObject("Scripting.Dictionary")
    Set periodos = CreateObject("Scripting.Dictionary")
    
    Dim r As Long
    For r = 1 To UBound(arr, 1)
        Dim codR As String, fDia As Variant, vVar As Variant
        codR = Trim(CStr(arr(r, 1)))
        fDia = arr(r, 3)
        vVar = arr(r, 4)
        
        If fondosPedidos.exists(codR) And IsDate(fDia) And IsNumeric(vVar) Then
            Dim periodo As String
            If tipo = "LOCAL" Then
                periodo = Format(CDate(fDia), "yyyy-mm")
            Else
                periodo = Year(CDate(fDia)) & "-Q" & ((Month(CDate(fDia)) - 1) \ 3 + 1)
            End If
            periodos(periodo) = 1
            
            Dim clave As String
            clave = codR & "|" & periodo
            
            Dim fNum As Double, varNum As Double
            fNum = CDbl(CDate(fDia))
            varNum = CDbl(vVar)
            
            If datos.exists(clave) Then
                Dim parts() As String
                parts = Split(datos(clave), "||")
                Dim fechaGuardada As Double, varGuardada As Double
                fechaGuardada = CDbl(parts(0))
                varGuardada = CDbl(parts(1))
                
                Dim reemplazar As Boolean: reemplazar = False
                If varGuardada = 0 And varNum <> 0 Then
                    reemplazar = True
                ElseIf varGuardada <> 0 And varNum = 0 Then
                    reemplazar = False
                Else
                    If fNum > fechaGuardada Then reemplazar = True
                End If
                
                If reemplazar Then
                    datos(clave) = CStr(fNum) & "||" & CStr(varNum)
                End If
            Else
                datos(clave) = CStr(fNum) & "||" & CStr(varNum)
            End If
        End If
    Next r
    
    If periodos.Count = 0 Then
        wsCuadro.Range("A1").Value = "No se encontraron datos para los códigos indicados en Resumen (" & tipo & ")."
        Exit Sub
    End If
    
    Dim listaPer() As String
    ReDim listaPer(0 To periodos.Count - 1)
    Dim idx2 As Long: idx2 = 0
    Dim k As Variant
    For Each k In periodos.keys
        listaPer(idx2) = CStr(k): idx2 = idx2 + 1
    Next k
    OrdenarArray listaPer
    
    wsCuadro.Range("A1").Value = "Nombre Fondo"
    wsCuadro.Range("B1").Value = "Código SBS"
    
    Dim c As Long
    For c = 0 To UBound(listaPer)
        Dim encab As String
        If tipo = "LOCAL" Then
            Dim pm() As String
            pm = Split(listaPer(c), "-")
            Dim fchTmp As Date
            fchTmp = DateSerial(CInt(pm(0)), CInt(pm(1)), 1)
            encab = Format(fchTmp, "mmm-yy")
            wsCuadro.Cells(1, 3 + c).NumberFormat = "@"
            wsCuadro.Cells(1, 3 + c).Value = encab
        Else
            Dim pq() As String
            pq = Split(listaPer(c), "-Q")
            encab = "Q" & pq(1) & "-" & pq(0)
            wsCuadro.Cells(1, 3 + c).NumberFormat = "@"
            wsCuadro.Cells(1, 3 + c).Value = encab
        End If
    Next c
    
    Dim fila As Long: fila = 2
    Dim kk As Variant
    For Each kk In ordenPedidos
        wsCuadro.Cells(fila, 1).Value = fondosPedidos(kk)
        wsCuadro.Cells(fila, 2).Value = kk
        
        For c = 0 To UBound(listaPer)
            Dim cl As String
            cl = kk & "|" & listaPer(c)
            If datos.exists(cl) Then
                Dim pp() As String
                pp = Split(datos(cl), "||")
                wsCuadro.Cells(fila, 3 + c).Value = CDbl(pp(1))
                wsCuadro.Cells(fila, 3 + c).NumberFormat = "0.00%"
            End If
        Next c
        
        fila = fila + 1
    Next kk
    
    With wsCuadro
        .Rows(1).Font.Bold = True
        .Range("A1:B1").Interior.Color = RGB(200, 215, 235)
        .Range(.Cells(1, 3), .Cells(1, 3 + UBound(listaPer))).Interior.Color = RGB(220, 230, 200)
        .Columns.AutoFit
    End With

End Sub

'==================================================================
' MACRO 3: Base 100 - 4 cuadros separados
'==================================================================
Sub GenerarBase100YGraficos()
    
    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    Dim wsCuadroLoc As Worksheet, wsCuadroExt As Worksheet, wsFondos As Worksheet, wsPesos As Worksheet
    
    On Error Resume Next
    Set wsCuadroLoc = wb.Sheets("Cuadro Locales")
    Set wsCuadroExt = wb.Sheets("Cuadro Extranjeros")
    Set wsFondos = wb.Sheets("Fondos")
    Set wsPesos = wb.Sheets("Pesos")
    On Error GoTo 0
    
    If wsCuadroLoc Is Nothing Or wsCuadroExt Is Nothing _
       Or wsFondos Is Nothing Or wsPesos Is Nothing Then
        MsgBox "Faltan hojas necesarias.", vbCritical
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    Dim tipos As Object: Set tipos = LeerTiposFondos(wsFondos)
    Dim origenes As Object: Set origenes = LeerOrigenFondos(wsFondos)
    Dim nombresFondos As Object: Set nombresFondos = LeerNombresFondos(wsFondos)
    Dim pesosF2 As Object: Set pesosF2 = LeerPesosDic25(wsPesos, "FONDO 2")
    Dim pesosF3 As Object: Set pesosF3 = LeerPesosDic25(wsPesos, "FONDO 3")
    
    Dim wsBase As Worksheet
    Set wsBase = ObtenerHoja(wb, "Base 100")
    wsBase.Cells.Clear
    
    Dim fila As Long: fila = 1
    fila = ArmarTablaBase100(wsBase, fila, wsCuadroLoc, tipos, origenes, nombresFondos, pesosF2, "F2 - LOCALES", "LOCAL")
    fila = fila + 2
    fila = ArmarTablaBase100(wsBase, fila, wsCuadroExt, tipos, origenes, nombresFondos, pesosF2, "F2 - EXTRANJEROS", "EXTRANJERO")
    fila = fila + 2
    fila = ArmarTablaBase100(wsBase, fila, wsCuadroLoc, tipos, origenes, nombresFondos, pesosF3, "F3 - LOCALES", "LOCAL")
    fila = fila + 2
    fila = ArmarTablaBase100(wsBase, fila, wsCuadroExt, tipos, origenes, nombresFondos, pesosF3, "F3 - EXTRANJEROS", "EXTRANJERO")
    
    wsBase.Columns.AutoFit
    
    Dim wsGraf As Worksheet
    Set wsGraf = ObtenerHoja(wb, "Gráficos")
    
    Dim ch As ChartObject
    For Each ch In wsGraf.ChartObjects
        ch.Delete
    Next ch
    wsGraf.Cells.Clear
    
    GenerarGraficosPorTipo wsBase, wsGraf, tipos
    
    Application.ScreenUpdating = True

End Sub

Private Function ArmarTablaBase100(wsDest As Worksheet, filaInicio As Long, _
                                    wsCuadro As Worksheet, tipos As Object, _
                                    origenes As Object, nombresFondos As Object, _
                                    pesos As Object, titulo As String, _
                                    origenFiltro As String) As Long
    
    Dim fila As Long: fila = filaInicio
    
    Dim ultColC As Long, ultFilaC As Long
    ultColC = wsCuadro.Cells(1, wsCuadro.Columns.Count).End(xlToLeft).Column
    ultFilaC = wsCuadro.Cells(wsCuadro.Rows.Count, 1).End(xlUp).Row
    
    Dim filaPorCodigo As Object
    Set filaPorCodigo = CreateObject("Scripting.Dictionary")
    Dim rr As Long
    For rr = 2 To ultFilaC
        Dim codCuadro As String
        codCuadro = Trim(CStr(wsCuadro.Cells(rr, 2).Value))
        If codCuadro <> "" Then filaPorCodigo(codCuadro) = rr
    Next rr
    
    Dim cols2026 As Collection, encabs2026 As Collection
    Set cols2026 = New Collection
    Set encabs2026 = New Collection
    
    Dim c As Long
    For c = 3 To ultColC
        Dim h As String
        h = Trim(CStr(wsCuadro.Cells(1, c).Value))
        If h <> "" Then
            Dim anio As Integer: anio = 0
            On Error Resume Next
            If InStr(LCase(h), "q") = 1 Then
                Dim pq() As String: pq = Split(h, "-")
                If UBound(pq) >= 1 Then anio = CInt(pq(1))
            Else
                Dim fch As Date
                fch = DateValue("1-" & h)
                If Err.Number = 0 Then anio = Year(fch)
            End If
            Err.Clear
            On Error GoTo 0
            If anio = 2026 Then
                cols2026.Add c
                encabs2026.Add h
            End If
        End If
    Next c
    
    If cols2026.Count = 0 Then
        wsDest.Cells(fila, 1).Value = titulo & " (no hay datos de 2026)"
        ArmarTablaBase100 = fila + 2
        Exit Function
    End If
    
    Dim ultColTot As Long: ultColTot = 5 + cols2026.Count
    
    wsDest.Cells(fila, 1).Value = titulo & " - Retorno Acumulado (Dic-25 = 0%)"
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Font.Size = 12
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColTot)).Interior.Color = RGB(180, 30, 30)
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColTot)).Font.Color = RGB(255, 255, 255)
    fila = fila + 1
    
    wsDest.Cells(fila, 1).Value = "Retornos 2026"
    wsDest.Cells(fila, 2).Value = "Código SBS"
    wsDest.Cells(fila, 3).Value = "Tipo"
    wsDest.Cells(fila, 4).Value = "Peso Dic-25"
    wsDest.Cells(fila, 5).Value = "Dic-25"
    Dim k As Long
    For k = 1 To encabs2026.Count
        wsDest.Cells(fila, 5 + k).Value = encabs2026(k)
    Next k
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColTot)).Font.Bold = True
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColTot)).Interior.Color = RGB(220, 230, 200)
    fila = fila + 1
    
    Dim codigoIter As Variant
    For Each codigoIter In pesos.keys
        Dim codigo As String: codigo = CStr(codigoIter)
        Dim peso As Double: peso = CDbl(pesos(codigo))
        
        Dim origenFondo As String
        If origenes.exists(codigo) Then
            origenFondo = CStr(origenes(codigo))
        Else
            origenFondo = "EXTRANJERO"
        End If
        If origenFondo <> origenFiltro Then GoTo SiguienteFondo
        
        Dim nombre As String
        If nombresFondos.exists(codigo) Then
            nombre = CStr(nombresFondos(codigo))
        Else
            nombre = codigo
        End If
        
        Dim tipoF As String: tipoF = "Sin clasificar"
        If tipos.exists(codigo) Then tipoF = CStr(tipos(codigo))
        
        wsDest.Cells(fila, 1).Value = nombre
        wsDest.Cells(fila, 2).Value = codigo
        wsDest.Cells(fila, 3).Value = tipoF
        wsDest.Cells(fila, 4).Value = peso
        wsDest.Cells(fila, 4).NumberFormat = "0.00%"
        wsDest.Cells(fila, 5).Value = 0
        wsDest.Cells(fila, 5).NumberFormat = "0.00%"
        
        Dim acumulado As Double: acumulado = 1
        If filaPorCodigo.exists(codigo) Then
            Dim filaC As Long: filaC = CLng(filaPorCodigo(codigo))
            Dim cc As Long
            For cc = 1 To cols2026.Count
                Dim colOrig As Long: colOrig = cols2026(cc)
                Dim vRet As Variant
                vRet = wsCuadro.Cells(filaC, colOrig).Value
                If IsNumeric(vRet) And vRet <> "" Then
                    acumulado = acumulado * (1 + CDbl(vRet))
                End If
                wsDest.Cells(fila, 5 + cc).Value = acumulado - 1
                wsDest.Cells(fila, 5 + cc).NumberFormat = "0.00%"
            Next cc
        Else
            For cc = 1 To cols2026.Count
                wsDest.Cells(fila, 5 + cc).Value = 0
                wsDest.Cells(fila, 5 + cc).NumberFormat = "0.00%"
            Next cc
        End If
        
        fila = fila + 1
SiguienteFondo:
    Next codigoIter
    
    ArmarTablaBase100 = fila
End Function

Private Sub GenerarGraficosPorTipo(wsBase As Worksheet, wsGraf As Worksheet, tipos As Object)
    
    Dim ultFila As Long
    ultFila = wsBase.Cells(wsBase.Rows.Count, 1).End(xlUp).Row
    
    Dim r As Long
    Dim posicionGrafico As Long: posicionGrafico = 1
    
    r = 1
    Do While r <= ultFila
        Dim titulo As String
        titulo = CStr(wsBase.Cells(r, 1).Value)
        
        If InStr(UCase(titulo), "RETORNO ACUMULADO") > 0 Then
            Dim filaEnc As Long: filaEnc = r + 1
            Dim filaInicioDatos As Long: filaInicioDatos = r + 2
            
            Dim ultCol As Long
            ultCol = wsBase.Cells(filaEnc, wsBase.Columns.Count).End(xlToLeft).Column
            
            Dim filaFinDatos As Long: filaFinDatos = filaInicioDatos
            Do While Trim(CStr(wsBase.Cells(filaFinDatos, 2).Value)) <> ""
                filaFinDatos = filaFinDatos + 1
                If filaFinDatos > ultFila Then Exit Do
            Loop
            filaFinDatos = filaFinDatos - 1
            
            If filaFinDatos >= filaInicioDatos Then
                Dim porTipo As Object
                Set porTipo = CreateObject("Scripting.Dictionary")
                Dim ordenTipos As Collection
                Set ordenTipos = New Collection
                
                Dim rrr As Long
                For rrr = filaInicioDatos To filaFinDatos
                    Dim tipoF As String
                    tipoF = Trim(CStr(wsBase.Cells(rrr, 3).Value))
                    If tipoF = "" Then tipoF = "Sin clasificar"
                    If Not porTipo.exists(tipoF) Then
                        Set porTipo(tipoF) = New Collection
                        ordenTipos.Add tipoF
                    End If
                    porTipo(tipoF).Add rrr
                Next rrr
                
                Dim tipoIter As Variant
                For Each tipoIter In ordenTipos
                    Dim listaFilas As Collection
                    Set listaFilas = porTipo(CStr(tipoIter))
                    
                    CrearGrafico wsBase, wsGraf, listaFilas, filaEnc, ultCol, _
                                 titulo & " - " & CStr(tipoIter), posicionGrafico
                    posicionGrafico = posicionGrafico + 1
                Next tipoIter
            End If
            
            r = filaFinDatos + 1
        Else
            r = r + 1
        End If
    Loop
End Sub

Private Sub CrearGrafico(wsBase As Worksheet, wsGraf As Worksheet, _
                          filasFondos As Collection, filaEncabezados As Long, _
                          ultCol As Long, titulo As String, posicion As Long)
    
    Dim colPos As Long, rowPos As Long
    colPos = ((posicion - 1) Mod 2) * 9 + 1
    rowPos = ((posicion - 1) \ 2) * 22 + 1
    
    Dim leftPos As Double, topPos As Double
    leftPos = wsGraf.Cells(rowPos, colPos).Left
    topPos = wsGraf.Cells(rowPos, colPos).Top
    
    Dim chartObj As ChartObject
    Set chartObj = wsGraf.ChartObjects.Add(Left:=leftPos, Top:=topPos, Width:=520, Height:=320)
    
    Dim cht As Chart
    Set cht = chartObj.Chart
    cht.ChartType = xlLine
    
    Do While cht.SeriesCollection.Count > 0
        cht.SeriesCollection(1).Delete
    Loop
    
    Dim colInicioDatos As Long: colInicioDatos = 5
    
    Dim item As Variant
    For Each item In filasFondos
        Dim filaFondo As Long: filaFondo = CLng(item)
        Dim nombreFondo As String
        nombreFondo = CStr(wsBase.Cells(filaFondo, 1).Value)
        
        Dim s As Series
        Set s = cht.SeriesCollection.NewSeries
        s.Name = nombreFondo
        s.Values = wsBase.Range(wsBase.Cells(filaFondo, colInicioDatos), wsBase.Cells(filaFondo, ultCol))
        s.XValues = wsBase.Range(wsBase.Cells(filaEncabezados, colInicioDatos), wsBase.Cells(filaEncabezados, ultCol))
        
        Dim ultPunto As Long
        ultPunto = ultCol - colInicioDatos + 1
        
        Dim p As Point
        Set p = s.Points(ultPunto)
        p.HasDataLabel = True
        p.DataLabel.Text = nombreFondo
        p.DataLabel.Position = xlLabelPositionRight
        p.DataLabel.Font.Size = 8
        p.DataLabel.Font.Bold = True
    Next item
    
    Dim soloTipo As String
    Dim posSep As Long: posSep = InStrRev(titulo, " - ")
    If posSep > 0 Then
        soloTipo = Mid(titulo, posSep + 3)
    Else
        soloTipo = titulo
    End If
    
    cht.HasTitle = True
    cht.ChartTitle.Text = soloTipo
    cht.ChartTitle.Font.Size = 14
    cht.ChartTitle.Font.Bold = True
    
    cht.HasLegend = True
    cht.Legend.Position = xlLegendPositionRight
    cht.Legend.Font.Size = 8
    
    On Error Resume Next
    cht.Axes(xlCategory).Crosses = xlMinimum
    cht.Axes(xlCategory).TickLabels.Orientation = -45
    cht.Axes(xlCategory).TickLabels.Font.Size = 8
    On Error GoTo 0
    
    On Error Resume Next
    cht.Axes(xlValue).HasMajorGridlines = True
    cht.Axes(xlValue).TickLabels.NumberFormat = "0.00%"
    cht.Axes(xlValue).TickLabels.Font.Size = 8
    On Error GoTo 0
    
End Sub

Private Function ArmarBloquePxR(wsDest As Worksheet, filaInicio As Long, _
                                  wsCuadro As Worksheet, tipos As Object, _
                                  vintages As Object, origenes As Object, _
                                  nombresFondos As Object, pesos As Object, _
                                  nombreFondo As String, origenFiltro As String) As Long
    
    Dim fila As Long: fila = filaInicio
    
    Dim ultColC As Long, ultFilaC As Long
    ultColC = wsCuadro.Cells(1, wsCuadro.Columns.Count).End(xlToLeft).Column
    ultFilaC = wsCuadro.Cells(wsCuadro.Rows.Count, 1).End(xlUp).Row
    
    Dim filaPorCodigo As Object
    Set filaPorCodigo = CreateObject("Scripting.Dictionary")
    Dim rr As Long
    For rr = 2 To ultFilaC
        Dim codCuadro As String
        codCuadro = Trim(CStr(wsCuadro.Cells(rr, 2).Value))
        If codCuadro <> "" Then filaPorCodigo(codCuadro) = rr
    Next rr
    
    Dim colsPer As Collection, encabsPer As Collection
    Set colsPer = New Collection
    Set encabsPer = New Collection
    
    Dim c As Long
    For c = 3 To ultColC
        Dim h As String
        h = Trim(CStr(wsCuadro.Cells(1, c).Value))
        If h <> "" Then
            Dim a As Integer: a = 0
            On Error Resume Next
            If InStr(LCase(h), "q") = 1 Then
                Dim pq() As String: pq = Split(h, "-")
                If UBound(pq) >= 1 Then a = CInt(pq(1))
            Else
                Dim fch As Date
                fch = DateValue("1-" & h)
                If Err.Number = 0 Then a = Year(fch)
            End If
            Err.Clear
            On Error GoTo 0
            If a = 2026 Then
                colsPer.Add c
                encabsPer.Add h
            End If
        End If
    Next c
    
    If colsPer.Count = 0 Then
        wsDest.Cells(fila, 1).Value = nombreFondo & " (sin datos 2026)"
        ArmarBloquePxR = fila + 1
        Exit Function
    End If
    
    wsDest.Cells(fila, 1).Value = nombreFondo
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Font.Size = 12
    fila = fila + 1
    
    wsDest.Cells(fila, 1).Value = "Retornos 2026"
    wsDest.Cells(fila, 2).Value = "Código SBS"
    wsDest.Cells(fila, 3).Value = "Tipo"
    wsDest.Cells(fila, 4).Value = "Vintage"
    wsDest.Cells(fila, 5).Value = "Dic-25"
    Dim k As Long
    For k = 1 To encabsPer.Count
        wsDest.Cells(fila, 5 + k).Value = encabsPer(k)
    Next k
    Dim ultColEnc As Long: ultColEnc = 5 + encabsPer.Count
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColEnc)).Font.Bold = True
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColEnc)).Interior.Color = RGB(180, 30, 30)
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColEnc)).Font.Color = RGB(255, 255, 255)
    fila = fila + 1
    
    Dim filaInicioFondos As Long: filaInicioFondos = fila
    
    Dim ordenTipos As Collection
    Set ordenTipos = New Collection
    Dim vistosTipo As Object
    Set vistosTipo = CreateObject("Scripting.Dictionary")
    
    Dim codigoIter As Variant
    For Each codigoIter In pesos.keys
        Dim codigo As String: codigo = CStr(codigoIter)
        Dim peso As Double: peso = CDbl(pesos(codigo))
        
        Dim origenFondo As String
        If origenes.exists(codigo) Then
            origenFondo = CStr(origenes(codigo))
        Else
            origenFondo = "EXTRANJERO"
        End If
        If origenFondo <> origenFiltro Then GoTo SiguienteFondo
        
        Dim nombre As String
        If nombresFondos.exists(codigo) Then
            nombre = CStr(nombresFondos(codigo))
        Else
            nombre = codigo
        End If
        
        Dim tipoF As String: tipoF = "Sin clasificar"
        If tipos.exists(codigo) Then tipoF = CStr(tipos(codigo))
        Dim vintage As String: vintage = ""
        If vintages.exists(codigo) Then vintage = CStr(vintages(codigo))
        
        If Not vistosTipo.exists(tipoF) Then
            vistosTipo(tipoF) = 1
            ordenTipos.Add tipoF
        End If
        
        wsDest.Cells(fila, 1).Value = nombre
        wsDest.Cells(fila, 2).Value = codigo
        wsDest.Cells(fila, 3).Value = tipoF
        wsDest.Cells(fila, 4).Value = vintage
        wsDest.Cells(fila, 5).Value = 0
        wsDest.Cells(fila, 5).NumberFormat = "0.00%"
        
        Dim cc As Long
        For cc = 1 To colsPer.Count
            Dim valor As Double: valor = 0
            If filaPorCodigo.exists(codigo) Then
                Dim filaC As Long: filaC = CLng(filaPorCodigo(codigo))
                Dim colOrig As Long: colOrig = colsPer(cc)
                Dim vRet As Variant
                vRet = wsCuadro.Cells(filaC, colOrig).Value
                If IsNumeric(vRet) And vRet <> "" Then
                    valor = CDbl(vRet) * peso
                End If
            End If
            wsDest.Cells(fila, 5 + cc).Value = valor
            wsDest.Cells(fila, 5 + cc).NumberFormat = "0.00%"
        Next cc
        
        fila = fila + 1
SiguienteFondo:
    Next codigoIter
    
    Dim filaFinFondos As Long: filaFinFondos = fila - 1
    
    If filaFinFondos < filaInicioFondos Then
        ArmarBloquePxR = fila
        Exit Function
    End If
    
    ' MTD (suma total)
    wsDest.Cells(fila, 1).Value = "MTD"
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 5).Formula = "=SUM(E" & filaInicioFondos & ":E" & filaFinFondos & ")"
    wsDest.Cells(fila, 5).NumberFormat = "0.00%"
    Dim col As Long
    For col = 1 To colsPer.Count
        Dim letraCol As String: letraCol = Columna(5 + col)
        wsDest.Cells(fila, 5 + col).Formula = "=SUM(" & letraCol & filaInicioFondos & ":" & letraCol & filaFinFondos & ")"
        wsDest.Cells(fila, 5 + col).NumberFormat = "0.00%"
    Next col
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColEnc)).Font.Bold = True
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, ultColEnc)).Interior.Color = RGB(220, 230, 200)
    fila = fila + 1
    
    ' Subtotales por tipo
    Dim tipoIter As Variant
    For Each tipoIter In ordenTipos
        wsDest.Cells(fila, 1).Value = CStr(tipoIter)
        wsDest.Cells(fila, 1).Font.Bold = True
        wsDest.Cells(fila, 5).Formula = "=SUMIF(C" & filaInicioFondos & ":C" & filaFinFondos & _
                                          ",""" & CStr(tipoIter) & """,E" & filaInicioFondos & ":E" & filaFinFondos & ")"
        wsDest.Cells(fila, 5).NumberFormat = "0.00%"
        For col = 1 To colsPer.Count
            Dim letraCol2 As String: letraCol2 = Columna(5 + col)
            wsDest.Cells(fila, 5 + col).Formula = "=SUMIF(C" & filaInicioFondos & ":C" & filaFinFondos & _
                                                    ",""" & CStr(tipoIter) & """," & letraCol2 & filaInicioFondos & ":" & letraCol2 & filaFinFondos & ")"
            wsDest.Cells(fila, 5 + col).NumberFormat = "0.00%"
        Next col
        fila = fila + 1
    Next tipoIter
    
    ArmarBloquePxR = fila
End Function

'==================================================================
' Arma un bloque del Resumen Total leyendo de Pesos x Retornos
'==================================================================
Private Function ArmarBloqueResumen(wsDest As Worksheet, filaInicio As Long, _
                                     wsPxR As Worksheet, _
                                     nombreFondoDisplay As String, _
                                     etiquetaFondoPxR As String) As Long
    
    Dim fila As Long: fila = filaInicio
    
    ' Localizar las dos secciones en Pesos x Retornos: LOCAL/etiquetaFondoPxR y EXTRANJERO/etiquetaFondoPxR
    Dim infoLoc As String, infoExt As String
    infoLoc = LocalizarSeccionPxR(wsPxR, "LOCAL", etiquetaFondoPxR)
    infoExt = LocalizarSeccionPxR(wsPxR, "EXTRANJERO", etiquetaFondoPxR)
    
    ' Parsing: "filaEnc|colDic25|colUltMes|filaMTD|filaPE|filaInfr|filaRE|filaPD|filaPESec|filaPEFoF"
    ' En realidad guardamos: "filaEnc|colDic25|colUltMes|filaMTD|tipos..."
    
    If infoLoc = "" And infoExt = "" Then
        wsDest.Cells(fila, 1).Value = nombreFondoDisplay & " (no se encontró sección)"
        ArmarBloqueResumen = fila + 1
        Exit Function
    End If
    
    ' Título FONDO X
    wsDest.Cells(fila, 1).Value = nombreFondoDisplay
    wsDest.Cells(fila, 1).Font.Bold = True
    wsDest.Cells(fila, 1).Font.Size = 14
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 3)).Interior.Color = RGB(150, 180, 220)
    fila = fila + 1
    
    wsDest.Cells(fila, 1).Value = "Categoría"
    wsDest.Cells(fila, 2).Value = "MTD"
    wsDest.Cells(fila, 3).Value = "YTD"
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 3)).Font.Bold = True
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 3)).Interior.Color = RGB(220, 230, 200)
    fila = fila + 1
    
    Dim filaLocalTotal As Long: filaLocalTotal = 0
    Dim filaExtTotal As Long: filaExtTotal = 0
    Dim filasSubtotalesLoc As Collection
    Dim filasSubtotalesExt As Collection
    Set filasSubtotalesLoc = New Collection
    Set filasSubtotalesExt = New Collection
    
    ' LOCAL
    If infoLoc <> "" Then
        filaLocalTotal = fila
        fila = fila + 1
        Dim primeraFilaLoc As Long: primeraFilaLoc = fila
        fila = EscribirTiposBloque(wsDest, fila, wsPxR, infoLoc, filasSubtotalesLoc)
        Dim ultimaFilaLoc As Long: ultimaFilaLoc = fila - 1
        
        wsDest.Cells(filaLocalTotal, 1).Value = "LOCAL TOTAL"
        If ultimaFilaLoc >= primeraFilaLoc Then
            wsDest.Cells(filaLocalTotal, 2).Formula = "=SUM(B" & primeraFilaLoc & ":B" & ultimaFilaLoc & ")"
            wsDest.Cells(filaLocalTotal, 3).Formula = "=SUM(C" & primeraFilaLoc & ":C" & ultimaFilaLoc & ")"
        End If
        wsDest.Cells(filaLocalTotal, 2).NumberFormat = "0.00%"
        wsDest.Cells(filaLocalTotal, 3).NumberFormat = "0.00%"
        wsDest.Range(wsDest.Cells(filaLocalTotal, 1), wsDest.Cells(filaLocalTotal, 3)).Font.Bold = True
        wsDest.Range(wsDest.Cells(filaLocalTotal, 1), wsDest.Cells(filaLocalTotal, 3)).Interior.Color = RGB(240, 240, 200)
        fila = fila + 1
    End If
    
    ' EXTRANJERO
    If infoExt <> "" Then
        filaExtTotal = fila
        fila = fila + 1
        Dim primeraFilaExt As Long: primeraFilaExt = fila
        fila = EscribirTiposBloque(wsDest, fila, wsPxR, infoExt, filasSubtotalesExt)
        Dim ultimaFilaExt As Long: ultimaFilaExt = fila - 1
        
        wsDest.Cells(filaExtTotal, 1).Value = "EXTRANJERO TOTAL"
        If ultimaFilaExt >= primeraFilaExt Then
            wsDest.Cells(filaExtTotal, 2).Formula = "=SUM(B" & primeraFilaExt & ":B" & ultimaFilaExt & ")"
            wsDest.Cells(filaExtTotal, 3).Formula = "=SUM(C" & primeraFilaExt & ":C" & ultimaFilaExt & ")"
        End If
        wsDest.Cells(filaExtTotal, 2).NumberFormat = "0.00%"
        wsDest.Cells(filaExtTotal, 3).NumberFormat = "0.00%"
        wsDest.Range(wsDest.Cells(filaExtTotal, 1), wsDest.Cells(filaExtTotal, 3)).Font.Bold = True
        wsDest.Range(wsDest.Cells(filaExtTotal, 1), wsDest.Cells(filaExtTotal, 3)).Interior.Color = RGB(240, 240, 200)
        fila = fila + 1
    End If
    
    ' TOTAL FONDO X = Local Total + Extranjero Total
    fila = fila + 1
    wsDest.Cells(fila, 1).Value = "TOTAL " & nombreFondoDisplay
    Dim formMtd As String, formYtd As String
    formMtd = "=0"
    formYtd = "=0"
    If filaLocalTotal > 0 And filaExtTotal > 0 Then
        formMtd = "=B" & filaLocalTotal & "+B" & filaExtTotal
        formYtd = "=C" & filaLocalTotal & "+C" & filaExtTotal
    ElseIf filaLocalTotal > 0 Then
        formMtd = "=B" & filaLocalTotal
        formYtd = "=C" & filaLocalTotal
    ElseIf filaExtTotal > 0 Then
        formMtd = "=B" & filaExtTotal
        formYtd = "=C" & filaExtTotal
    End If
    wsDest.Cells(fila, 2).Formula = formMtd
    wsDest.Cells(fila, 2).NumberFormat = "0.00%"
    wsDest.Cells(fila, 3).Formula = formYtd
    wsDest.Cells(fila, 3).NumberFormat = "0.00%"
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 3)).Font.Bold = True
    wsDest.Range(wsDest.Cells(fila, 1), wsDest.Cells(fila, 3)).Interior.Color = RGB(180, 200, 230)
    fila = fila + 1
    
    ArmarBloqueResumen = fila
End Function

'==================================================================
' Localiza la sección "LOCAL/Fondo X" o "EXTRANJERO/Fondo X" en Pesos x Retornos
' Devuelve: "filaEnc|colDic25|colUltMes|filaMTD|tipos_filas..."
' Si no encuentra, devuelve ""
'==================================================================
Private Function LocalizarSeccionPxR(wsPxR As Worksheet, _
                                      origenBuscar As String, _
                                      fondoBuscar As String) As String
    
    Dim ultFila As Long
    ultFila = wsPxR.Cells(wsPxR.Rows.Count, 1).End(xlUp).Row
    
    ' Buscar la sección de origen (LOCAL o EXTRANJERO)
    Dim filaOrigen As Long: filaOrigen = 0
    Dim r As Long
    For r = 1 To ultFila
        If UCase(Trim(CStr(wsPxR.Cells(r, 1).Value))) = UCase(origenBuscar) Then
            filaOrigen = r
            Exit For
        End If
    Next r
    
    If filaOrigen = 0 Then
        LocalizarSeccionPxR = ""
        Exit Function
    End If
    
    ' Buscar dónde termina la sección (siguiente "LOCAL" o "EXTRANJERO")
    Dim filaFinSeccion As Long: filaFinSeccion = ultFila
    For r = filaOrigen + 1 To ultFila
        Dim val As String: val = UCase(Trim(CStr(wsPxR.Cells(r, 1).Value)))
        If val = "LOCAL" Or val = "EXTRANJERO" Then
            filaFinSeccion = r - 1
            Exit For
        End If
    Next r
    
    ' Buscar "Fondo 2" o "Fondo 3" dentro de esa sección
    Dim filaFondo As Long: filaFondo = 0
    For r = filaOrigen + 1 To filaFinSeccion
        If LCase(Trim(CStr(wsPxR.Cells(r, 1).Value))) = LCase(fondoBuscar) Then
            filaFondo = r
            Exit For
        End If
    Next r
    
    If filaFondo = 0 Then
        LocalizarSeccionPxR = ""
        Exit Function
    End If
    
    ' Fila encabezados = filaFondo + 1
    Dim filaEnc As Long: filaEnc = filaFondo + 1
    
    ' Detectar colDic25 y colUltMes
    Dim ultCol As Long
    ultCol = wsPxR.Cells(filaEnc, wsPxR.Columns.Count).End(xlToLeft).Column
    
    Dim colDic25 As Long: colDic25 = 0
    Dim colUltMes As Long: colUltMes = ultCol
    Dim c As Long
    For c = 5 To ultCol
        Dim h As String: h = LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value)))
        If h = "dic-25" Or h = "dic25" Then colDic25 = c
    Next c
    
    If colDic25 = 0 Then colDic25 = 5
    
    ' Buscar fila "MTD" después de los fondos
    Dim filaMTD As Long: filaMTD = 0
    For r = filaEnc + 1 To filaFinSeccion
        If UCase(Trim(CStr(wsPxR.Cells(r, 1).Value))) = "MTD" Then
            filaMTD = r
            Exit For
        End If
    Next r
    
    If filaMTD = 0 Then
        LocalizarSeccionPxR = ""
        Exit Function
    End If
    
    ' Recolectar filas de subtotales por tipo (después de MTD, hasta línea vacía o siguiente FondoX)
    Dim tiposInfo As String: tiposInfo = ""
    For r = filaMTD + 1 To filaFinSeccion
        Dim etiqueta As String: etiqueta = Trim(CStr(wsPxR.Cells(r, 1).Value))
        If etiqueta = "" Then Exit For
        If LCase(etiqueta) = "fondo 2" Or LCase(etiqueta) = "fondo 3" Then Exit For
        tiposInfo = tiposInfo & ";" & etiqueta & ":" & r
    Next r
    
    LocalizarSeccionPxR = filaEnc & "|" & colDic25 & "|" & colUltMes & "|" & filaMTD & tiposInfo
End Function

'==================================================================
' Escribe los tipos en el Resumen leyendo de los subtotales de Pesos x Retornos
'==================================================================
Private Function EscribirTiposBloque(wsDest As Worksheet, filaInicio As Long, _
                                      wsPxR As Worksheet, info As String, _
                                      filasOut As Collection) As Long
    Dim fila As Long: fila = filaInicio
    
    Dim partes() As String
    partes = Split(info, "|")
    Dim filaEnc As Long: filaEnc = CLng(partes(0))
    Dim colDic25 As Long: colDic25 = CLng(partes(1))
    Dim colUltMes As Long: colUltMes = CLng(partes(2))
    
    Dim subpartes() As String
    subpartes = Split(partes(3), ";")
    
    ' Leer mes seleccionado de B1
    Dim mesSeleccionado As String
    mesSeleccionado = Trim(CStr(wsDest.Range("B1").Value))
    
    ' Buscar la columna del mes seleccionado en los encabezados de PxR
    Dim colMesSel As Long: colMesSel = 0
    Dim c As Long
    For c = colDic25 To colUltMes
        If LCase(Trim(CStr(wsPxR.Cells(filaEnc, c).Value))) = LCase(mesSeleccionado) Then
            colMesSel = c
            Exit For
        End If
    Next c
    
    Dim primerMesCol As Long: primerMesCol = colDic25 + 1
    
    Dim i As Long
    For i = 1 To UBound(subpartes)
        Dim subItem As String: subItem = subpartes(i)
        If subItem <> "" Then
            Dim sp() As String
            sp = Split(subItem, ":")
            If UBound(sp) >= 1 Then
                Dim tipo As String: tipo = sp(0)
                Dim filaTipoPxR As Long: filaTipoPxR = CLng(sp(1))
                
                wsDest.Cells(fila, 1).Value = "  " & tipo
                
                ' MTD: valor del mes seleccionado
                Dim mtd As Double: mtd = 0
                If colMesSel > 0 Then
                    Dim vMtd As Variant
                    vMtd = wsPxR.Cells(filaTipoPxR, colMesSel).Value
                    If IsNumeric(vMtd) Then mtd = CDbl(vMtd)
                End If
                wsDest.Cells(fila, 2).Value = mtd
                wsDest.Cells(fila, 2).NumberFormat = "0.00%"
                
                ' YTD: producto compuesto desde primerMesCol hasta colMesSel
                Dim ytd As Double: ytd = 0
                If colMesSel > 0 And colMesSel >= primerMesCol Then
                    Dim acumulado As Double: acumulado = 1
                    Dim cc As Long
                    For cc = primerMesCol To colMesSel
                        Dim vRet As Variant
                        vRet = wsPxR.Cells(filaTipoPxR, cc).Value
                        If IsNumeric(vRet) Then
                            acumulado = acumulado * (1 + CDbl(vRet))
                        End If
                    Next cc
                    ytd = acumulado - 1
                End If
                wsDest.Cells(fila, 3).Value = ytd
                wsDest.Cells(fila, 3).NumberFormat = "0.00%"
                
                fila = fila + 1
            End If
        End If
    Next i
    
    EscribirTiposBloque = fila
End Function

Private Function LeerPesosDic25(ws As Worksheet, etiqueta As String) As Object
    
    Dim dic As Object
    Set dic = CreateObject("Scripting.Dictionary")
    
    Dim ultCol As Long, ultFila As Long
    ultCol = 50
    ultFila = 300
    
    ' PASO 1: Buscar "% de Cartera" para saltar bloque de montos
    Dim filaInicioBusqueda As Long: filaInicioBusqueda = 1
    
    Dim r As Long, c As Long
    For r = 1 To ultFila
        For c = 1 To ultCol
            On Error Resume Next
            Dim valTxt As String
            valTxt = LCase(Trim(CStr(ws.Cells(r, c).Value)))
            On Error GoTo 0
            If valTxt = "% de cartera" Or valTxt = "%decartera" Then
                filaInicioBusqueda = r
                Exit For
            End If
        Next c
        If filaInicioBusqueda > 1 Then Exit For
    Next r
    
    If filaInicioBusqueda = 1 Then filaInicioBusqueda = 70
    
    ' PASO 2: Buscar etiqueta FONDO X
    Dim filaEtiqueta As Long: filaEtiqueta = 0
    Dim colEtiqueta As Long: colEtiqueta = 0
    
    For r = filaInicioBusqueda To ultFila
        For c = 1 To ultCol
            On Error Resume Next
            If UCase(Trim(CStr(ws.Cells(r, c).Value))) = UCase(etiqueta) Then
                filaEtiqueta = r
                colEtiqueta = c
            End If
            On Error GoTo 0
            If filaEtiqueta > 0 Then Exit For
        Next c
        If filaEtiqueta > 0 Then Exit For
    Next r
    
    If filaEtiqueta = 0 Then Set LeerPesosDic25 = dic: Exit Function
    
    ' PASO 3: Detectar fin del bloque
    Dim colFinBloque As Long: colFinBloque = ultCol
    Dim rEnc As Long
    For rEnc = filaEtiqueta To filaEtiqueta + 3
        For c = colEtiqueta + 1 To ultCol
            On Error Resume Next
            Dim val0 As String
            val0 = UCase(Trim(CStr(ws.Cells(rEnc, c).Value)))
            On Error GoTo 0
            If Left(val0, 6) = "FONDO " And val0 <> UCase(etiqueta) Then
                colFinBloque = c - 1
                Exit For
            End If
        Next c
        If colFinBloque < ultCol Then Exit For
    Next rEnc
    
    ' PASO 4: Buscar Código SBS y Dic-25
    Dim filaEnc As Long: filaEnc = 0
    Dim colCod As Long: colCod = 0
    Dim colDic As Long: colDic = 0
    
    Dim rFin As Long
    rFin = filaEtiqueta + 10
    If rFin > ultFila Then rFin = ultFila
    
    For r = filaEtiqueta To rFin
        Dim tmpCod As Long, tmpDic As Long
        tmpCod = 0: tmpDic = 0
        
        For c = colEtiqueta To colFinBloque
            On Error Resume Next
            Dim valRaw As Variant
            valRaw = ws.Cells(r, c).Value
            
            Dim valTxt2 As String
            valTxt2 = LCase(Replace(Trim(CStr(valRaw)), " ", ""))
            If valTxt2 = "códigosbs" Or valTxt2 = "codigosbs" Then tmpCod = c
            
            If valTxt2 = "dic-25" Or valTxt2 = "dic25" Then tmpDic = c
            If IsDate(valRaw) Then
                If Year(CDate(valRaw)) = 2025 And Month(CDate(valRaw)) = 12 Then tmpDic = c
            End If
            
            On Error GoTo 0
        Next c
        
        If tmpCod > 0 And tmpDic > 0 Then
            colCod = tmpCod
            colDic = tmpDic
            filaEnc = r
            Exit For
        End If
    Next r
    
    If colCod = 0 Or colDic = 0 Then Set LeerPesosDic25 = dic: Exit Function
    
    ' PASO 5: Leer fondos
    Dim i As Long
    For i = filaEnc + 1 To ultFila
        On Error Resume Next
        Dim cod As String, pesoVal As Variant
        cod = Trim(CStr(ws.Cells(i, colCod).Value))
        
        If UCase(Left(cod, 6)) = "FONDO " Then Exit For
        If cod = "" Then Exit For
        
        pesoVal = ws.Cells(i, colDic).Value
        If IsNumeric(pesoVal) Then
            dic(cod) = CDbl(pesoVal)
        Else
            dic(cod) = 0
        End If
        On Error GoTo 0
    Next i
    
    Set LeerPesosDic25 = dic
End Function

Private Sub OrdenarArray(arr() As String)
    Dim i As Long, j As Long, tmp As String
    For i = LBound(arr) To UBound(arr) - 1
        For j = i + 1 To UBound(arr)
            If arr(i) > arr(j) Then
                tmp = arr(i): arr(i) = arr(j): arr(j) = tmp
            End If
        Next j
    Next i
End Sub

Private Function Columna(num As Long) As String
    Dim resultado As String
    resultado = ""
    Do While num > 0
        Dim resto As Long
        resto = (num - 1) Mod 26
        resultado = Chr(65 + resto) & resultado
        num = (num - resto - 1) \ 26
    Loop
    Columna = resultado
End Function

                Dim tipo As String: tipo = sp(0)
                Dim filaTipoPxR As Long: filaTipoPxR = CLng(sp(1))
                msj = msj & "  '" & tipo & "' -> fila " & filaTipoPxR & vbCrLf
                
                ' Probar leer un valor de esa fila para confirmar
                Dim valPrueba As Variant
                valPrueba = wsPxR.Cells(filaTipoPxR, primerMesCol).Value
                msj = msj & "    Valor en " & Columna(primerMesCol) & filaTipoPxR & ": " & valPrueba & vbCrLf
            End If
        End If
    Next i
    
    MsgBox msj, vbInformation, "Debug YTD"
